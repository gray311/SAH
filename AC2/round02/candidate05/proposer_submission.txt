    implementation_py: |
      def run(ctx, args):
          import random
          num_variants = args.get("num_variants", 5)
          rep = args.get("representation", "piecewise-constant")
          n = int(ctx.get_program().split("num_intervals")[1].split(" = ")[1].strip()) if "num_intervals" in ctx.get_program() else 300
          
          variants = []
          for i in range(num_variants):
              key = jax.random.PRNGKey(i + ctx.budget_left()[0] * 100)
              f = jnp.zeros(n)
              
              if rep == "piecewise-constant":
                  start = int(random.uniform(0.1, 0.4) * n)
                  width = int(random.uniform(0.3, 0.6) * n)
                  height = random.uniform(0.5, 2.0)
                  f = f.at[start:start+width].set(height)
              elif rep == "gaussian":
                  K = random.randint(2, 5)
                  for k in range(K):
                      mu = int(random.uniform(0.15, 0.85) * n)
                      sigma = int(random.uniform(10, 50))
                      weight = random.uniform(0.5, 2.0)
                      f = f + weight * jnp.exp(-0.5 * ((jnp.arange(n) - mu) / sigma)**2)
              elif rep == "piecewise-linear":
                  num_nodes = random.randint(n//2, n)
                  nodes = sorted(random.sample(range(n), num_nodes))
                  values = [random.uniform(0.5, 2.0) for _ in range(num_nodes)]
                  for idx, (node, val) in enumerate(zip(nodes, values)):
                      start_idx = nodes[idx-1] if idx > 0 else 0
                      end_idx = nodes[idx+1] if idx < len(nodes)-1 else n
                      f = f.at[start_idx:end_idx].set(jnp.linspace(val, values[(idx+1)%len(values)], end_idx-start_idx+1))
              elif rep == "multiscale":
                  f = jnp.zeros(n)
                  coarse_start = int(random.uniform(0.2, 0.4) * n)
                  coarse_width = int(random.uniform(0.3, 0.5) * n)
                  f = f.at[coarse_start:coarse_start+coarse_width].set(1.5)
                  fine_start = int(random.uniform(0.25, 0.6) * n)
                  fine_width = int(random.uniform(0.2, 0.4) * n)
                  f = f.at[fine_start:fine_start+fine_width].set(2.0)
              
              variants.append({"variant": i, "f_values": f.tolist()})
          
          return {"num_variants": num_variants, "representation": rep, "variants": variants}
