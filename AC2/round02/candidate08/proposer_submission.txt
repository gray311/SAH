      def run(ctx, args):
          import jax
          import jax.numpy as jnp
          results = []
          n = 300

          # Step functions
          f1 = jnp.zeros(n); f1 = f1.at[int(0.2*n):int(0.8*n)].set(1.0)
          results.append(("step", "wide_sym", f1))
          f2 = jnp.zeros(n); f2 = f2.at[int(0.3*n):int(0.7*n)].set(1.2)
          results.append(("step", "narrow_sym", f2))
          f3 = jnp.zeros(n); f3 = f3.at[int(0.15*n):int(0.55*n)].set(1.1)
          results.append(("step", "asym_left", f3))
          f4 = jnp.zeros(n)
          f4 = f4.at[int(0.1*n):int(0.25*n)].set(0.8)
          f4 = f4.at[int(0.25*n):int(0.6*n)].set(1.5)
          f4 = f4.at[int(0.6*n):int(0.9*n)].set(0.7)
          results.append(("step", "multi_level", f4))

          # Piecewise-linear
          f5 = jnp.zeros(100)
          f5 = jnp.concatenate([jnp.zeros(25), jnp.ones(50), jnp.zeros(25)])
          results.append(("pwlinear", "100int", f5))
          f6 = jnp.zeros(200)
          f6 = jnp.concatenate([jnp.linspace(0, 2, 100), jnp.linspace(2, 0, 100)])
          results.append(("pwlinear", "200int_tri", f6))
          f7 = jnp.zeros(300)
          f7 = jnp.concatenate([jnp.zeros(50), jnp.ones(50), jnp.zeros(50), jnp.ones(50), jnp.zeros(50), jnp.ones(50), jnp.zeros(50)])
          results.append(("pwlinear", "300int_multi", f7))

          # Gaussian mixtures
          def make_gauss(mus, sigmas, weights):
              f = jnp.zeros(n)
              for mu, sigma, w in zip(mus, sigmas, weights):
                  f = f + w * jnp.exp(-((jnp.arange(n) - mu * n) / (sigma * n))**2)
              return f
          f8 = make_gauss([0.5, 0.5], [0.2, 0.2], [0.5, 0.5])
          results.append(("gaussian", "2gauss", f8))
          f9 = make_gauss([0.33, 0.5, 0.67], [0.15, 0.15, 0.15], [0.33, 0.34, 0.33])
          results.append(("gaussian", "3gauss", f9))
          f10 = make_gauss([0.2, 0.3, 0.5, 0.7, 0.8], [0.05]*5, [0.2]*5)
          results.append(("gaussian", "5gauss_narrow", f10))

          # Exponential
          def make_exp(mus, alphas, weights):
              f = jnp.zeros(n)
              for mu, alpha, w in zip(mus, alphas, weights):
                  f = f + w * jnp.exp(-alpha * jnp.abs((jnp.arange(n) - mu * n) / n))
              return f
          f11 = make_exp([0.5], [1.0], [1.0])
          results.append(("exp", "single", f11))
          f12 = make_exp([0.3, 0.7], [0.8, 0.5], [0.5, 0.5])
          results.append(("exp", "double", f12))

          # Probe all
          for repr_name, variant_name, f_vals in results:
              padded = jnp.pad(jnp.maximum(0, f_vals), (0, n))
              key = jax.random.PRNGKey(42)
              try:
                  probe_score = ctx.probe()
                  results[-1] = (repr_name, variant_name, float(probe_score))
              except:
                  results[-1] = (repr_name, variant_name, -999)

          results_sorted = sorted(results, key=lambda x: x[2], reverse=True)
          return {"representations": results_sorted}

    hook: before_model
    description: |
      Intervenes when agent stays in one representation too long. Checks last 5 iterations; if same class 4+ times, reminds to try different class.
    implementation_py: |
      def before_model(hook_input):
          state = hook_input.get("state", {})
          recent_reps = state.get("recent_representations", [])
          iterations = state.get("iteration", 0)

          if iterations >= 5 and len(recent_reps) >= 5:
              last_repr = recent_reps[-1] if recent_reps else "unknown"
              same_count = sum(1 for r in recent_reps[-5:] if r == last_repr)
              if same_count >= 4:
                  return "DIVERSITY_WARNING: Used {} for {} consecutive iterations. Likely local optimum. Use scan_representations to probe other classes or try variants of different class.".format(last_repr, same_count)
          return None
