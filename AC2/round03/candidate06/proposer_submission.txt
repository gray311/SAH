  description: Middleware that forces switching to a different function family after 3 evaluations on the same family without improvement, preventing over-tuning of single representations.
  implementation_py: |
    def before_model(hook_input):
        state = hook_input.get('state', {})
        iteration = state.get('iteration', 0)
        family_history = state.get('family_history', [])
        evals_on_family = state.get('evals_on_family', {})
        
        if len(evals_on_family) > 0:
            current_family = family_history[-1] if family_history else None
            evals_count = evals_on_family.get(current_family, 0) if current_family else 0
            
            if current_family is not None and evals_count >= 3:
                available_families = ['step', 'triangle', 'trapezoid', 'gaussian', 'exponential', 'bspline']
                used_families = [fh.split(':')[0].split('_')[0] for fh in family_history]
                unused_families = [f for f in available_families if f not in used_families]
                
                if unused_families:
                    next_family = unused_families[0]
                    return "FAMILY_SWITCH_RECOMMENDED: Evaluated %s %d times. Switch to %s templates. Call code_template_probe for %s variants." % (current_family, evals_count, next_family, next_family)
                else:
                    return "FAMILY_SWITCH_RECOMMENDED: Exhausted all families. Re-examine your parameter variations within current family or try combining templates."
        
        return None
