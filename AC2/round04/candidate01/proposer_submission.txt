    implementation_py: |
      def run(ctx, args):
          num_vars = args.get("num_variants", 8)
          base_w = args.get("base_width", 0.5)
          num_seg = args.get("num_segments", 3)
          
          variants = []
          N = ctx.get_program()
          
          for i in range(num_vars):
              segs = num_seg
              if i % 4 == 0 and segs > 2:
                  segs = 2
              elif i % 4 == 1 and segs < 5:
                  segs = 5
              
              width = base_w * (0.7 + 0.6 * (i % 5) / 5.0)
              start_pos = int((1.0 - width) / 2.0 * N)
              end_pos = int((1.0 + width) / 2.0 * N)
              h1 = 1.0
              heights = [h1]
              
              if segs == 2:
                  h2 = 1.0 + 0.3 * (i % 10 + 1)
                  heights = [h1, h2]
              elif segs == 3:
                  h2 = 1.0 + 0.2 * (i % 8 + 1)
                  h3 = 1.0 + 0.15 * ((i + 2) % 8 + 1)
                  heights = [h1, h2, h3]
              else:
                  h2 = 1.0 + 0.2 * (i % 8 + 1)
                  h3 = 1.0 + 0.15 * ((i + 2) % 8 + 1)
                  h4 = 1.0 + 0.1 * ((i + 3) % 8 + 1)
                  if segs == 4:
                      heights = [h1, h2, h3]
                  else:
                      heights = [h1, h2, h3, h4]
              
              segment_size = max(1, (end_pos - start_pos) // (len(heights) - 1)) if len(heights) > 1 else 1
              height_str = ",".join(str(h) for h in heights)
              offsets = [start_pos + seg * segment_size for seg in range(len(heights))]
              offset_val = ",".join(str(o) for o in offsets)
              
              heights_last = heights[-1]
              variant_code = '''# Step-function variant
N = {num_intervals}
f = jnp.zeros(N)
heights = [{height_str}]
for seg_idx, height in enumerate(heights):
    start_offset = {offset_val}[seg_idx]
    end_offset = {offset_val}[seg_idx + 1] if seg_idx + 1 < len({offset_val}) else N
    f = f.at[start_offset:end_offset].set({last_height})
'''.format(num_intervals=num_intervals, height_str=height_str, offset_val=offset_val, last_height=heights_last)
              
              variants.append({"variant_id": i, "code": variant_code})
          
          return {"variants": variants, "N": num_intervals}
