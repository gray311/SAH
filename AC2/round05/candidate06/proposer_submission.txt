    input_schema:

    implementation_py: |
      def run(ctx, args):
          f_values = ctx.get_best_program()
          if f_values is None:
              return {"note": "No function yet. Edit first."}
          import jax.numpy as jnp
          h = 1.0 / (len(f_values) + 1)
          convolution = jnp.fft.ifft(jnp.fft.fft(f_values) * jnp.fft.fft(f_values)).real
          peak_idx = jnp.argmax(jnp.abs(convolution))
          peak_val = jnp.abs(convolution[peak_idx])
          tail_region = jnp.abs(convolution[len(convolution)//4:])
          tail_mean = jnp.mean(tail_region)
          tail_max = jnp.max(tail_region)
          discretization_error_est = h * tail_mean * len(tail_region)
          if discretization_error_est > 0.01:
              rec = "Try finer discretization (increase num_intervals)"
          else:
              rec = "Optimize learning rate or step count"
          return {
              "num_intervals": 400,
              "peak_location": int(peak_idx),
              "peak_value": float(peak_val),
              "tail_mean": float(tail_mean),
              "tail_max": float(tail_max),
              "discretization_error_estimate": float(discretization_error_est),
              "recommendation": rec
          }
