      def run(ctx, args):
          import jax
          import jax.numpy as jnp
          f_values = ctx.get_program()
          if not f_values:
              return {"error": "No code"}
          try:
              f_arr = jnp.array([float(x) for x in f_values.strip().split('\n') if x.strip()])
              if len(f_arr) < 100:
                  return {"error": "Not enough points", "note": "Need at least 100 points for coarse grid"}
              N_coarse = 100
              h = 1.0 / N_coarse
              f_coarse = f_arr[:N_coarse]
              f_non_neg = jax.nn.relu(f_coarse)
              padded_f = jnp.pad(f_non_neg, (0, N_coarse))
              fft_f = jnp.fft.fft(padded_f)
              conv = jnp.fft.ifft(fft_f * fft_f).real
              
              num_conv = len(conv)
              y_points = jnp.concatenate([jnp.array([0.0]), conv, jnp.array([0.0])])

              norm1 = jnp.sum(jnp.abs(conv)) / (len(conv) + 1)
              norm_inf = jnp.max(jnp.abs(conv))
              denom = norm1 * norm_inf
              c2 = l2_sq / denom if denom > 0 else 0.0
              
              best_known = 0.8962799441554086
              combined = c2 / best_known
              
              return {"estimated_C2": float(c2), "coarse_grid_score": float(combined), "time_ms": 0}
          except Exception as e:
              return {"error": str(e), "note": "Probe failed"}
