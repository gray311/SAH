      def run(ctx, args):
          import random
          random.seed(hash(ctx.get_program()) % 2**31)
          
          n = 400
          f = jnp.zeros(n)
          
          # Novel pattern generation strategies
          strategy = random.choice(["multi_level", "asymmetric", "bimodal", "extreme_contrast", "plateau"])
          
          if strategy == "multi_level":
              # 5-7 levels with varied heights
              num_levels = random.randint(5, 7)
              starts = []
              ends = []
              for i in range(num_levels):
                  if i == 0:
                      starts.append(0)
                  else:
                      starts.append(int((i-0.15) * n / (num_levels - 0.5)))
                  ends.append(int((i+0.15) * n / (num_levels - 0.5)))
              heights = [round(random.uniform(0.8, 2.5), 2) for _ in range(num_levels)]
              for i in range(num_levels):
                  f = f.at[starts[i]:ends[i]].set(heights[i])
                  
          elif strategy == "asymmetric":
              # Left-heavy or right-heavy distribution
              split = random.choice([0.3, 0.4, 0.6, 0.7])
              # Left side: taller, narrower
              left_start = 0
              left_end = int(split * n)
              left_h = random.uniform(1.8, 2.4)
              # Right side: shorter, wider
              right_start = int(split * n)
              right_end = n
              right_h = random.uniform(0.8, 1.2)
              f = f.at[left_start:left_end].set(left_h)
              f = f.at[right_start:right_end].set(right_h)
              
          elif strategy == "bimodal":
              # Two peaks with valley between
              valley = random.uniform(0.25, 0.75)
              peak1_start, peak1_end = 0, int(valley * n * 0.4)
              peak2_start, peak2_end = int(valley * n * 0.6), n
              valley_start = peak1_end
              valley_end = peak2_start
              f = f.at[peak1_start:peak1_end].set(random.uniform(2.0, 2.6))
              f = f.at[peak2_start:peak2_end].set(random.uniform(2.0, 2.6))
              f = f.at[valley_start:valley_end].set(random.uniform(0.5, 0.9))
              
          elif strategy == "extreme_contrast":
              # Very high narrow peak with low wings
              peak_start = int(0.28 * n)
              peak_end = int(0.72 * n)
              peak_h = random.uniform(2.3, 2.8)
              wing_start = 0
              wing_end = peak_start
              wing_h = random.uniform(0.6, 0.9)
              wing2_start = peak_end
              wing2_end = n
              f = f.at[wing_start:wing_end].set(wing_h)
              f = f.at[peak_start:peak_end].set(peak_h)
              f = f.at[wing2_start:wing2_end].set(wing_h)
              
          elif strategy == "plateau":
              # Flat top with gradual edges
              flat_start = int(0.15 * n)
              flat_end = int(0.85 * n)
              flat_h = random.uniform(1.6, 2.0)
              left_ramp = random.uniform(0.7, 1.0)
              right_ramp = random.uniform(0.7, 1.0)
              f = f.at[0:flat_start].set(left_ramp)
              f = f.at[flat_start:flat_end].set(flat_h)
              f = f.at[flat_end:n].set(right_ramp)

                  "pattern_type": strategy, "num_levels": num_levels}
