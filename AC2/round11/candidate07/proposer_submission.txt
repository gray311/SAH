    description: |
      Fast approximate C2 scoring. Computes ||f★f||₂² / ((∫f)²||f★f||∞) on a subsampled grid (80 points) in ~10ms.
      Uses separate budget of 30 probes (independent of 30 full evaluations). Scores are approximate but preserve
      relative ordering well enough to filter 10-20 candidates. Call c2_probe on multiple variants, keep the top 2-3,
      then call evaluate_solution on those. This enables rapid exploration that full evaluations alone cannot support.
      Not ideal for: Final certification (use evaluate_solution for that).
    input_schema:
      type: object
      properties: {}
    implementation_py: |
      def run(ctx, args):
          import re
          import jax
          import jax.numpy as jnp
          prog = ctx.get_program()
          if "# EVOLVE-BLOCK" not in prog:
              return {"note": "no evolve block", "score": 0.0}
          try:
              num_intervals_match = re.search(r'num_intervals:\s*(\d+)', prog)
              if num_intervals_match:
                  N = int(num_intervals_match.group(1))
              else:
                  N = 200
              f_values = jnp.zeros(N)
              height_matches = re.findall(r'\.set\(([0-9.]+)\)', prog)
              if height_matches:
                  heights = [float(h) for h in height_matches]
                  avg_h = jnp.mean(jnp.array(heights))
                  step_start = int(0.25 * N)
                  step_end = int(0.75 * N)
                  f_values = f_values.at[step_start:step_end].set(avg_h)
              f_non_negative = jnp.maximum(f_values, 0)
              padded_f = jnp.pad(f_non_negative, (0, N))
              fft_f = jnp.fft.fft(padded_f)
              convolution = jnp.fft.ifft(fft_f * fft_f).real
              num_probe_points = 80
              probe_indices = jnp.linspace(0, len(convolution)-1, num_probe_points, dtype=int)
              conv_probe = convolution[probe_indices]
              h = 1.0 / (num_probe_points + 1)
              y_points = jnp.concatenate([jnp.array([0.0]), conv_probe, jnp.array([0.0])])
              y1, y2 = y_points[:-1], y_points[1:]
              l2_norm_squared = jnp.sum((h / 3) * (y1**2 + y1 * y2 + y2**2))
              norm_1 = jnp.sum(jnp.abs(conv_probe)) / (len(conv_probe) + 1)
              norm_inf = jnp.max(jnp.abs(conv_probe))
              denominator = norm_1 * norm_inf
              if denominator > 0:
                  c2_score = l2_norm_squared / denominator
              else:
                  c2_score = 0.0
              return {"score": float(c2_score), "probe_grid": num_probe_points}
          except Exception as e:
              return {"note": f"probe error: {str(e)}", "score": 0.0}
