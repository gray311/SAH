    improvement stalls to diagnose why the pattern needs refinement.

  implementation_py: |
    def run(ctx, args):
        import re
        prog = ctx.get_program()
        if "# EVOLVE-BLOCK" not in prog:
            return {"note": "no evolve block", "analysis": {}}
        
        height_matches = re.findall(r'\.set\((\d+\.\d+)\)', prog)
        height_matches += re.findall(r'set\((\d+\.\d+)\)', prog)
        heights = [float(h) for h in height_matches if h and re.match(r'^\d+\.\d+$', h)]
        
        if not heights:
            return {"note": "could not parse heights", "analysis": {}}
        
        avg_h = sum(heights) / len(heights)
        std_h = (sum((h - avg_h)**2 for h in heights) / len(heights)) ** 0.5
        min_h = min(heights)
        max_h = max(heights)
        h_range = max_h - min_h
        h_variance = std_h / avg_h if avg_h > 0 else 0
        
        num_levels = len(heights)
        pattern_type = "unknown"
        if h_range < 0.1:
            pattern_type = "nearly_uniform"
        elif h_range < 0.3:
            pattern_type = "moderate_variation"
        else:
            pattern_type = "high_variation"
        
        start_matches = re.findall(r'at\((\d+).*\.set', prog)
        start_matches += re.findall(r'at\((\d+),', prog)
        if start_matches:
            start_positions = [int(s) for s in start_matches[:5]]
            if start_positions:
                avg_start = sum(start_positions) / len(start_positions)
                center_of_mass = avg_start / (prog.count('at(') + 1)
            else:
                center_of_mass = 0.5
        else:
            center_of_mass = 0.5
        
        interval_count = len(re.findall(r'at\(', prog))
        
        return {
            "analysis": {
                "num_levels": num_levels,
                "avg_height": round(avg_h, 4),
                "std_height": round(std_h, 4),
                "min_height": round(min_h, 4),
                "max_height": round(max_h, 4),
                "height_range": round(h_range, 4),
                "height_variance": round(h_variance, 4),
                "pattern_type": pattern_type,
                "center_of_mass": round(center_of_mass, 4),
                "interval_count": interval_count,
                "num_proposals_recommended": 5 if pattern_type == "high_variation" else 3
            },
            "diagnosis": {
                "is_optimized": h_variance < 0.05,
                "recommendation": "If is_optimized=True and no improvement for 2 iterations, try new architectures"
            }
        }

    Separate budget of 30 probes (does not consume real evaluation budget).
    Scores are approximate but RELIABLE for this task - use to rapidly rank variants

  implementation_py: |
    def run(ctx, args):
        import numpy as np
        prog = ctx.get_program()
        if "# EVOLVE-BLOCK" not in prog:
            return {"note": "no evolve block", "c2_approx": 0.0}
        
        try:
            N_full = 600
            N_sample = int(N_full * 0.1)
            
            f_values = np.zeros(N_sample)
            
            height_matches = re.findall(r'\.set\((\d+\.\d+)\)', prog)
            height_matches += re.findall(r'set\((\d+\.\d+)\)', prog)
            heights = [float(h) for h in height_matches if h and re.match(r'^\d+\.\d+$', h)]
            
            if not heights:
                return {"note": "no pattern found", "c2_approx": 0.0}
            
            interval_matches = re.findall(r'at\((\d+),\s*(\d+)', prog)
            intervals = [(int(s), int(e)) for s, e in interval_matches]
            
            if intervals:
                for start, end in intervals:
                    start_idx = int(start * N_sample / 600)
                    end_idx = int(end * N_sample / 600)
                    f_values[start_idx:end_idx] = np.mean(heights) if len(heights) > 1 else float(heights[0])
            
            f_non_negative = np.maximum(f_values, 0)
            N = N_sample
            padded_f = np.pad(f_non_negative, (0, N))
            
            fft_f = np.fft.fft(padded_f)
            convolution = np.fft.ifft(fft_f * fft_f).real
            
            num_conv_points = len(convolution)
            h = 1.0 / (num_conv_points + 1)
            y_points = np.concatenate([np.array([0.0]), convolution, np.array([0.0])])
            y1, y2 = y_points[:-1], y_points[1:]
            l2_norm_squared = np.sum((h / 3) * (y1**2 + y1 * y2 + y2**2))
            
            norm_1 = np.sum(np.abs(convolution)) / (len(convolution) + 1)
            norm_inf = np.max(np.abs(convolution))
            denominator = norm_1 * norm_inf
            c2_ratio = l2_norm_squared / denominator if denominator > 0 else 0
            
            return {"c2_approx": round(c2_ratio, 6), "note": "probe result"}
        except Exception as e:
            return {"note": f"probe failed: {str(e)}", "c2_approx": 0.0}
