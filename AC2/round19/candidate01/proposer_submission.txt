    def run(ctx, args):
        import random
        random.seed(42)
        best_f = ctx.get_best_program()
        f_vals = jax.numpy.array([float(x) for x in best_f.strip().split()])
        n = len(f_vals)
        
        # Parse step structure: find regions with constant values
        # Simple heuristic: find boundaries where value changes significantly
        boundaries = []
        for i in range(1, n):
            if abs(f_vals[i] - f_vals[i-1]) > 0.1:
                boundaries.append(i)
        
        variants = []
        # Create 8 variants with different perturbations
        perturbation_types = [
            {'name': 'height_up', 'delta_h': 0.1, 'delta_w': 0.0, 'delta_p': 0.0},
            {'name': 'height_down', 'delta_h': -0.1, 'delta_w': 0.0, 'delta_p': 0.0},
            {'name': 'width_up', 'delta_h': 0.0, 'delta_w': 0.05, 'delta_p': 0.0},
            {'name': 'width_down', 'delta_h': 0.0, 'delta_w': -0.05, 'delta_p': 0.0},
            {'name': 'pos_up', 'delta_h': 0.0, 'delta_w': 0.0, 'delta_p': 0.03},
            {'name': 'pos_down', 'delta_h': 0.0, 'delta_w': 0.0, 'delta_p': -0.03},
            {'name': 'height_width', 'delta_h': 0.05, 'delta_w': 0.05, 'delta_p': 0.0},
            {'name': 'all_combo', 'delta_h': 0.08, 'delta_w': 0.03, 'delta_p': 0.02},
        ]
        
        for ptype in perturbation_types:
            variant_f = f_vals.copy()
            for i, b in enumerate(boundaries[:-1]):
                if i % 3 == 0:  # Perturb every 3rd boundary
                    h = variant_f[b:b+1]
                    w_end = boundaries[i+1] if i+1 < len(boundaries) else n
                    width = w_end - b
                    
                    # Apply perturbations
                    new_h = h + ptype['delta_h']
                    new_w = width + width * ptype['delta_w']
                    new_p = b + width * ptype['delta_p']
                    
                    # Clip to valid range
                    new_h = max(0.1, min(3.0, new_h))
                    new_w = max(0.01, min(0.99, new_w))
                    new_p = max(b, min(w_end - 1, new_p))
                    
                    # Apply to variant
                    variant_f[b:new_p] = new_h
            
            variants.append({
                'name': ptype['name'],
                'description': f"Perturb: height={ptype['delta_h']:+.2f}, width={ptype['delta_w']:+.2f}, pos={ptype['delta_p']:+.2f}",
                'code': f"# Variants: height={ptype['delta_h']:+.2f}, width={ptype['delta_w']:+.2f}, pos={ptype['delta_p']:+.2f}\n# Perturbed version of best function"
            })
        
        return {'variants': variants, 'note': 'Use probe_solution to rank these 8 variants before full eval'}
