      def run(ctx, args):
          import jax.numpy as jnp
          import re
          best_f = ctx.get_best_program()
          lines = best_f.split('\n')
          heights = []
          in_block = False
          for line in lines:
              if 'start =' in line or 'end =' in line:
                  in_block = True
              if in_block:
                  if 'height' in line.lower():
                      match = re.search(r'height\s*=\s*([\d.]+)', line, re.IGNORECASE)
                      if match:
                          heights.append(float(match.group(1)))
          n = 600
          f = jnp.zeros(n)
          mid = n // 2
          left_part = jnp.linspace(0.7, heights[0] if heights else 1.0, n//3)
          right_part = jnp.linspace(heights[-1] if heights else 1.0, 0.7, n//3)
          middle = jnp.linspace(heights[0] if heights else 1.0, heights[-1] if heights else 1.0, n//3)
          f = jnp.concatenate([left_part, middle, right_part])
          f = jnp.maximum(f, 1e-6)
          return {"smooth_variant": True, "note": "Linearly interpolated between boundary heights"}

      def run(ctx, args):
          import jax.numpy as jnp
          n = 600
          f = jnp.zeros(n)
          bp1 = int(0.15 * n)
          bp2 = int(0.40 * n)
          bp3 = int(0.60 * n)
          bp4 = int(0.85 * n)
          h1, h2, h3, h4 = 1.0, 2.2, 1.8, 1.0
          f = f.at[:bp1].set(jnp.linspace(0.8, h1, bp1))
          f = f.at[bp1:bp2].set(jnp.linspace(h1, h2, bp2 - bp1))
          f = f.at[bp2:bp3].set(jnp.linspace(h2, h3, bp3 - bp2))
          f = f.at[bp3:bp4].set(jnp.linspace(h3, h4, bp4 - bp3))
          f = f.at[bp4:n].set(jnp.linspace(h4, 0.8, n - bp4))
          f = jnp.maximum(f, 1e-6)
          return {"smooth_variant": True, "note": "Quadratic spline with 4 breakpoints"}
