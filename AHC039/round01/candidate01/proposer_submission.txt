      Analyze the spatial distribution of all fish. Returns: total mackerels/sardines,
      global bounding box, density heatmap (50x50 grid cell counts), and per-region
      mackerel/sardine ratios. Call this ONCE at the start of your program to

    input_schema: {type: object, properties: {}}
    implementation_py: |
      def run(ctx, args):
          # Read all fish coordinates
          all_fish = []
          names = ctx.list_task_inputs()
          if names:
              df = ctx.read_input_df(names[0], nrows=5000)
              if len(df) >= 5000:
                  mackerels = df.iloc[:5000]
                  sardines = df.iloc[5000:]
                  for _, row in mackerels.iterrows():
                      all_fish.append({"type": 1, "x": int(row[0]), "y": int(row[1])})
                  for _, row in sardines.iterrows():
                      all_fish.append({"type": -1, "x": int(row[0]), "y": int(row[1])})
          
          if not all_fish:
              return {"error": "No fish found", "mackerels": 0, "sardines": 0,
                      "bbox_xmin": 0, "bbox_ymin": 0, "bbox_xmax": 100000, "bbox_ymax": 100000}
          
          # Compute bounding box
          xs = [f["x"] for f in all_fish]
          ys = [f["y"] for f in all_fish]
          xmin, xmax = min(xs), max(xs)
          ymin, ymax = min(ys), max(ys)
          
          # Build 50x50 grid density heatmap
          grid_size = 50
          cell_width = (xmax - xmin) / grid_size if xmax > xmin else 1
          cell_height = (ymax - ymin) / grid_size if ymax > ymin else 1
          
          grid = [[{"mackerel": 0, "sardine": 0} for _ in range(grid_size)] for _ in range(grid_size)]
          for f in all_fish:
              cx = int((f["x"] - xmin) / cell_width) if cell_width > 0 else 0
              cy = int((f["y"] - ymin) / cell_height) if cell_height > 0 else 0
              if 0 <= cx < grid_size and 0 <= cy < grid_size:
                  if f["type"] == 1:
                      grid[cy][cx]["mackerel"] += 1
                  else:
                      grid[cy][cx]["sardine"] += 1
          
          mackerels = sum(f["type"] == 1 for f in all_fish)
          sardines = sum(f["type"] == -1 for f in all_fish)
          
          return {
              "mackerels": mackerels, "sardines": sardines,
              "bbox_xmin": xmin, "bbox_ymin": ymin, "bbox_xmax": xmax, "bbox_ymax": ymax,
              "grid": grid,
              "recommendation": "Use greedy clustering around mackerel cells with high mackerel/sardine ratio"
          }
