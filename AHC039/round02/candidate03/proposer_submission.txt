          max_y = max(p[1] for p in mackerels)
          
          # Count points in bounding box
          m_count = 0
          s_count = 0
          for p in mackerels:
              if min_x <= p[0] <= max_x and min_y <= p[1] <= max_y:
                  m_count += 1
          for p in sardines:
