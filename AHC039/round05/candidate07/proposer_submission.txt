      def run(ctx, args):
          vertices = args.get("vertices", [])
          if len(vertices) < 4 or len(vertices) > 1000:
              return {"valid": False, "errors": ["bad count"], "perimeter": 0}
          p = 0
          for i, v in enumerate(vertices):
              x, y = v.get("x", 0), v.get("y", 0)
              if not (0 <= x <= 100000 and 0 <= y <= 100000):
                  return {"valid": False, "errors": ["out of range"], "perimeter": 0}
              prev = vertices[(i-1) % len(vertices)]
              p += abs(x - prev.get("x", 0)) + abs(y - prev.get("y", 0))
          if p >= 400000:
              return {"valid": False, "errors": ["perimeter too large"], "perimeter": p}
          return {"valid": True, "perimeter": p, "errors": []}
