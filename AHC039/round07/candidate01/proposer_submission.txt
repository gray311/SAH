    description: |
      Analyze the mackerel positions to find dense clusters. Returns top clusters
      with their centroids, densities, and distances to nearest sardines. This
      guides polygon construction to target high-value regions first. Call ONCE
      at the start of search to inform all subsequent polygon designs.
    input_schema:
      type: object
      properties: {}
    implementation_py: |
      def run(ctx, args):
          # Get all fish positions
          mackerels = ctx.read_input_df("mackerels.csv", nrows=5000)
          sardines = ctx.read_input_df("sardines.csv", nrows=5000)
          
          if mackerels.empty:
              return {"note": "No mackerel data"}
          
          # Cluster analysis: use simple density-based clustering
          # Compute grid cells and count mackerels per cell
          cell_size = 5000  # 5km grid cells
          grid = {}
          for idx, row in mackerels.iterrows():
              cell_x = row['x'] // cell_size
              cell_y = row['y'] // cell_size
              key = (cell_x, cell_y)
              grid[key] = grid.get(key, 0) + 1
          
          # Find top clusters by density
          cluster_info = []
          for (cx, cy), count in sorted(grid.items(), key=lambda x: -x[1])[:10]:
              # Get centroid of mackerels in this cluster
              cluster_mask = (mackerels['x'] >= cx * cell_size) & \
                            (mackerels['x'] < (cx + 1) * cell_size) & \
                            (mackerels['y'] >= cy * cell_size) & \
                            (mackerels['y'] < (cy + 1) * cell_size)
              cluster_mackerels = mackerels[cluster_mask]
              
              if not cluster_mackerels.empty:
                  centroid_x = int(cluster_mackerels['x'].mean())
                  centroid_y = int(cluster_mackerels['y'].mean())
                  
                  # Find nearest sardine
                  min_sardine_dist = float('inf')
                  for idx, sr in sardines.iterrows():
                      dist = ((sr['x'] - centroid_x)**2 + **(sr['y'] - centroid_y)2)**0.5
                      min_sardine_dist = min(min_sardine_dist, dist)
                  
                  cluster_info.append({
                      "centroid_x": centroid_x,
                      "centroid_y": centroid_y,
                      "count": int(count),
                      "density": count / (cell_size ** 2),
                      "nearest_sardine_dist": float(min_sardine_dist)
                  })
          
          return {
              "num_clusters": len(cluster_info),
              "clusters": cluster_info
          }

      def run(ctx, args):
          min_x = args.get("min_x", 0)
          max_x = args.get("max_x", 100000)
          min_y = args.get("min_y", 0)
          max_y = args.get("max_y", 100000)
          
          # Quick grid-based approximation (sample every 10th point)
          mackerels = ctx.read_input_df("mackerels.csv", nrows=5000)
          sardines = ctx.read_input_df("sardines.csv", nrows=5000)
          
          # Approximate counts by subsampling
          sample_step = 50
          m_count = 0
          s_count = 0
          
          for idx in range(0, len(mackerels), sample_step):
              if mackerels.iloc[idx]['x'] >= min_x and mackerels.iloc[idx]['x'] <= max_x and \
                 mackerels.iloc[idx]['y'] >= min_y and mackerels.iloc[idx]['y'] <= max_y:
                  m_count += 1
          
          for idx in range(0, len(sardines), sample_step):
              if sardines.iloc[idx]['x'] >= min_x and sardines.iloc[idx]['x'] <= max_x and \
                 sardines.iloc[idx]['y'] >= min_y and sardines.iloc[idx]['y'] <= max_y:
                  s_count += 1
          
          # Scale back to estimate full counts
          scale_factor = len(mackerels) / sample_step
          est_m = int(m_count * scale_factor)
          est_s = int(s_count * scale_factor)
          
          return {
              "approx_mackerels": est_m,
              "approx_sardines": est_s,
              "approx_score": max(0, est_m - est_s + 1)
          }
