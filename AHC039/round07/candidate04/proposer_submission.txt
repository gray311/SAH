      Ideal for: comparing candidate regions, deciding whether to bite out sardine clusters,
      choosing between multiple small vs one large polygon.
    input_schema:

          type: integer

          type: integer
        maxY:
          type: integer
    implementation_py: |
      def run(ctx, args):
          min_x = args.get("minX", 0)
          max_x = args.get("maxX", 100000)
          min_y = args.get("minY", 0)
          max_y = args.get("maxY", 100000)
          
          # Count mackerels and sardines in region by sampling
          sample_mackerels = 0
          sample_sardines = 0
          
          x = min_x
          while x <= max_x:
              y = min_y
              while y <= max_y:
                  for px in range(x, min(x + 200, max_x + 1), 50):
                      for py in range(y, min(y + 200, max_y + 1), 50):

                                  fx, fy = fish["position"]
                                  if min_x <= fx <= max_x and min_y <= fy <= max_y:
                                      if i < 5000:
                                          sample_mackerels += 1
                                      else:
                                          sample_sardines += 1
                  y += 200
              x += 200
          
          net = sample_mackerels - sample_sardines
          ratio = sample_mackerels / max(sample_sardines, 1)
          
          return {
              "mackerels": sample_mackerels,
              "sardines": sample_sardines,
              "net_score": net,
              "density_ratio": round(ratio, 4)
          }
