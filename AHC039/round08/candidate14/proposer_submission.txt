        grid_size = 100
        grid = [[{'m': 0, 's': 0} for _ in range(grid_size)] for _ in range(grid_size)]
        
        for fish in ctx.get_program().strip().split('\n'):
            if 'fish[' not in fish:
                continue

                parts = fish.replace('fish[', '').replace(']', '').split(',')
                if len(parts) >= 2:
                    x, y, fish_type = int(parts[0]), int(parts[1]), 1 if 'mackerel' in fish else -1
                    cx, cy = x // cell_size, y // cell_size
                    if 0 <= cx < grid_size and 0 <= cy < grid_size:
                        grid[cy][cx]['m' if fish_type == 1 else 's'] += 1
            except:
                continue
        
        seed = args.get('seed_cluster', {})
        seed_cx, seed_cy = seed.get('cx', 50), seed.get('cy', 50)
        seed_m, seed_s = seed.get('m_count', 0), seed.get('s_count', 0)
        
        # Define bounding box around seed cluster
        min_x, max_x = max(0, seed_cx * cell_size - 200), min(100000, seed_cx * cell_size + 200)
        min_y, max_y = max(0, seed_cy * cell_size - 200), min(100000, seed_cy * cell_size + 200)
        
        # Family A: Multi-component - approximate by checking nearby clusters
        score_a = seed_m - seed_s
        # Check adjacent cells for additional mackerels
        for dcx in range(-2, 3):

                cx, cy = seed_cx + dcx, seed_cy + dcy
                if 0 <= cx < grid_size and 0 <= cy < grid_size:
                    m, s = grid[cy][cx]['m'], grid[cy][cx]['s']
                    score_a += m - s
        
        # Family B: Spiral - approximate by excluding sardines in extended region
        score_b = seed_m - seed_s
        for dcx in range(-3, 4):
            for dcy in range(-3, 4):
                cx, cy = seed_cx + dcx, seed_cy + dcy
                if 0 <= cx < grid_size and 0 <= cy < grid_size:
                    m, s = grid[cy][cx]['m'], grid[cy][cx]['s']
                    score_b += m - 0.5 * s  # Spiral may exclude some sardines

                score_c += max(0, m - s)  # Adaptive growth
