    def run(ctx, args):
        max_cells = args.get("max_cells", 50)
        program_text = ctx.get_program()
        fish_data = []
        for line in program_text.split('\n'):
            line = line.strip()
            if line.startswith('fish['):
                try:
                    content = line.replace('fish[', '').replace(']', '')
                    if ',mackerel' in content:
                        parts = content.replace(',mackerel', '').split(',')
                        x, y = int(parts[0]), int(parts[1])
                        fish_data.append((x, y, 1))
                    elif ',sardine' in content:
                        parts = content.replace(',sardine', '').split(',')
                        x, y = int(parts[0]), int(parts[1])
                        fish_data.append((x, y, -1))
                except:
                    continue
        
        grids = [
            (100, 100, 1000),
            (200, 200, 500),
            (500, 500, 200),
        ]
        
        grid_count = [[[(0, 0) for _ in range(g[1])] for _ in range(g[0])] for g in grids]
        
        for x, y, ftype in fish_data:
            for g_idx, (gx, gy, _) in enumerate(grids):
                cx, cy = x // gx, y // gy
                if 0 <= cx < gx and 0 <= cy < gy:
                    grid_count[g_idx][cy][cx] = (1, ftype) if ftype == 1 else (0, ftype)
        
        robust_cells = []
        base_size = 500
        for r in range(base_size):
            for c in range(base_size):
                cells_info = []
                for g_idx, (gx, gy, _) in enumerate(grids):
                    r_g, c_g = r // (gy/g_idx*100), c // (cx/g_idx*100) if g_idx > 0 else r, c
                    if r_g < gy and c_g < gx:
                        m, s = grid_count[g_idx][r_g][c_g]
                        cells_info.append((m, s, r_g, c_g))
                
                if len(cells_info) < 3:
                    continue
                
                positive_count = sum(1 for m, s, _, _ in cells_info if m >= s)
                if positive_count >= 2:
                    total_m = sum(m for m, s, _, _ in cells_info)
                    total_s = sum(s for m, s, _, _ in cells_info)
                    robust_cells.append({
                        "row": r,
                        "col": c,
                        "m": total_m,
                        "s": total_s,
                        "score": total_m - total_s,
                        "grids_covered": positive_count
                    })
        
        robust_cells.sort(key=lambda x: x["score"], reverse=True)
        return {
            "top_cells": robust_cells[:max_cells],
            "total_fish": len(fish_data),
            "grids_built": len(grids)
        }

    ## Detailed Method

    ### Step 1: Multi-Resolution Density Estimation

    Build three grids over [0,100000]x[0,100000]:
