    def run(ctx, args):
        import math
        import random
        
        poly = args.get("polygon", {})
        vertices = poly.get("vertices", [])
        subsample_ratio = args.get("subsample_ratio", 0.1)
        
        if not vertices:
            return {"error": "no polygon", "mackerels": 0, "sardines": 0}
        
        program = ctx.get_program()
        fish_list = []
        for line in program.split('\n'):
            line = line.strip()
            if 'mackerel' in line.lower():
                try:
                    coords = line.split(',')
                    if len(coords) >= 2:
                        x = int(coords[0].strip())
                        y = int(coords[1].strip())
                        fish_list.append((x, y, 1))
                except:
                    pass
            elif 'sardine' in line.lower():
                try:
                    coords = line.split(',')
                    if len(coords) >= 2:
                        x = int(coords[0].strip())
                        y = int(coords[1].strip())
                        fish_list.append((x, y, -1))
                except:
                    pass
        
        if not fish_list:
            return {"error": "no fish found", "mackerels": 0, "sardines": 0}
        
        n_subsample = max(1, int(len(fish_list) * subsample_ratio))
        shuffled = list(range(n_subsample))
        random.seed(42)
        for i in range(n_subsample - 1, 0, -1):
            j = random.randint(0, i)
            shuffled[i], shuffled[j] = shuffled[j], shuffled[i]
        sub_sample_indices = shuffled[:n_subsample]
        
        def point_in_polygon(px, py, poly):
            n = len(poly)
            inside = False
            j = n - 1
            for i in range(n):
                xi, yi = poly[i]
                xj, yj = poly[j]
                if ((yi > py) != (yj > py)) and (px < (xj - xi) * (py - yi) / (yj - yi) + xi):
                    inside = not inside
                j = i
            return inside
        
        mackerels = 0
        sardines = 0
        total_inside = 0
        
        for idx in sub_sample_indices:
            fx, fy, ftype = fish_list[idx]
            if point_in_polygon(fx, fy, vertices):
                total_inside += 1
                if ftype == 1:
                    mackerels += 1
                else:
                    sardines += 1
        
        total_est = len(fish_list)
        mack_est = int(mackerels * total_est / max(1, total_inside)) if total_inside > 0 else 0
        sard_est = int(sardines * total_est / max(1, total_inside)) if total_inside > 0 else 0
        
        score = max(0, mack_est - sard_est + 1)
        
        return {
            "subsampled_mackerels": mackerels,
            "subsampled_sardines": sardines,
            "estimated_mackerels": mack_est,
            "estimated_sardines": sard_est,
            "estimated_score": score,
            "subsample_count": len(sub_sample_indices),
            "total_fish": len(fish_list)
        }
