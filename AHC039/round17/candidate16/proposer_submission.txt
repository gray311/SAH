    def run(ctx, args):

        my = args.get("my", 0)
        
        best_rect = None
        best_score = -float('inf')
        
        # Try rectangles of size up to 3x3 centered near (mx, my)
        for size in [1, 2, 3]:
            for dx in range(-size+1, size):
                for dy in range(-size+1, size):
