      def run(ctx, args):
          import numpy as np
          variants = args.get("variants", [])
          if len(variants) < 2:
              return {"ranking": [], "note": "need at least 2 variants"}
          max_samples = args.get("max_samples", 5000)
          results = []
          for v in variants:
              name = v.get("name", "unnamed")
              code = v.get("code", "")
              try:
                  # Execute variant code in memory only (no edits to main program)
                  local_ns = {"torch": torch, "np": np}
                  exec(code, local_ns)
                  # If the variant defines rebalance_experts_hierarchical or balanced_packing or replicate_experts
                  variant_funcs = [k for k in local_ns.keys() if k.startswith(('rebalance','balanced','replicate'))]
                  if not variant_funcs:
                      results.append({"name": name, "score": -1, "error": "No variant function found"})
                      continue
                  # Pick the main rebalance function
                  main_func = None
                  for f in ['rebalance_experts_hierarchical', 'balanced_packing', 'replicate_experts']:
                      if f in local_ns:
                          main_func = local_ns[f]
                          break
                  if main_func is None:
                      results.append({"name": name, "score": -1, "error": "No recognized function found"})
                      continue
                  # Create a mock weight tensor (use first 20 experts for speed)
                  num_layers = 4
                  num_logical = min(64, max(8, max_samples // 2))
                  weight = torch.randn(num_layers, num_logical, dtype=torch.float32)
                  # Simple load balance heuristic: variance of row sums
                  def quick_eval(weight):
                      n, m = weight.shape
                      # Simple load balance heuristic: variance of row sums
                      flat = weight.flatten()
                      row_sums = flat[:n].view(-1, m).sum(dim=1)
                      if row_sums.numel() == 0:
                          return 0.0
                      return float(np.var(row_sums) / (row_sums.sum() ** 2 + 1e-8))
                  score = quick_eval(weight)
                  results.append({"name": name, "score": 1.0 - score})  # Lower variance = higher score
              except Exception as e:
                  results.append({"name": name, "score": -1, "error": str(e)})
          # Sort by score (higher is better)
          results.sort(key=lambda x: x["score"], reverse=True)
          rankings = [r["name"] for r in results]
          return {"ranking": rankings, "scores": {r["name"]: r["score"] for r in results}, "note": "ranked by load balance (higher is better)"}
