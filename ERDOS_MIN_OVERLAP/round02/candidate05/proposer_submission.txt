      def run(ctx, args):
          import jax
          import jax.numpy as jnp
          N = 800
          domain = 2.0
          dx = domain / N
          key = jax.random.PRNGKey(42)

          def sigmoid(z):
              return jax.nn.sigmoid(z)

          def create_bimodal(pos, width=0.3):
              x = jnp.linspace(0, domain, N)
              h = sigmoid(-5.0 * (x - pos) / width + 3.0)
              integral = jnp.sum(h) * dx
              h = h / integral
              return h

          def create_uniform():
              key, sub = jax.random.split(key)
              h = jax.random.uniform(sub, (N,), 0.3, 0.7)
              integral = jnp.sum(h) * dx
              h = h / integral
              return h

          def create_alternating():
              x = jnp.linspace(0, domain, N)
              h = (jnp.sin(2 * jnp.pi * x) + 2.0) / 3.0
              integral = jnp.sum(h) * dx
              h = h / integral
              return h

          def create_shifted(offset):
              x = jnp.linspace(0, domain, N)
              h = sigmoid(-6.0 * (x - offset) / 0.5 + 4.0)
              integral = jnp.sum(h) * dx
              h = h / integral
              return h

          results = {}
          # Bimodal at different positions
          for pos in [0.2, 0.8]:
              h = create_bimodal(pos)
              results[f'bimodal_pos{pos:.1f}'] = h
          # Uniform
          h = create_uniform()
              results['uniform'] = h
          # Alternating
          h = create_alternating()
              results['alternating'] = h
          # Shifted
          for off in [0.3, 0.7]:
              h = create_shifted(off)
              results[f'shifted{off:.1f}'] = h

          # Small perturbations of best patterns
          key, sub = jax.random.split(key)
          best_keys = ['bimodal_pos0.2', 'shifted0.3', 'shifted0.7', 'uniform']
          for k in best_keys[:4]:
              h = results[k] + jax.random.normal(sub, (N,)) * 0.1
              integral = jnp.sum(h) * dx
              h = h / integral
              results[f'{k}_pert'] = h

          return {'variants': results}
