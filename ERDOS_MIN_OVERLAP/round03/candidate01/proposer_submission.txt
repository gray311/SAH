      def run(ctx, args):
          import numpy as np
          N = 800
          domain = 2.0
          dx = domain / N
          candidates = []
          
          # Generate 3-step functions
          for _ in range(12):
              a = np.random.uniform(0.2, 0.8)
              b = np.random.uniform(a + 0.1, min(a + 1.0, 1.8))
              h1, h2, h3 = np.random.uniform(0.3, 0.7, 3)
              areas = np.array([a, b-a, 2-b])
              norm = h1*areas[0] + h2*areas[1] + h3*areas[2]
              if norm == 0: continue
              h1, h2, h3 = h1/norm, h2/norm, h3/norm
              
              step_vals = [h1, h2, h3]
              thresholds = [0, a, b, 2]
              
              x = np.linspace(0, domain, N)
              latent = np.zeros(N)
              latent = np.where((x >= 0) & (x < thresholds[1]), h1, latent)
              latent = np.where((x >= thresholds[1]) & (x < thresholds[2]), h2, latent)
              latent = np.where((x >= thresholds[2]) & (x < thresholds[3]), h3, latent)
              latent = np.clip(latent, 0, 1)
              integral = np.sum(latent) * dx
              if abs(integral - 1.0) > 0.01:
                  scale = 1.0 / integral
                  latent = latent * scale
                  latent = np.clip(latent, 0, 1)
              
              candidates.append({
                  "num_steps": 3,
                  "thresholds": [round(t, 6) for t in thresholds[:-1]],
                  "step_values": [round(float(v), 6) for v in step_vals],
                  "latent": latent.tolist(),
                  "approx_integral": round(integral, 6)
              })
          
          # Generate 4-step functions
          for _ in range(8):

              latent = np.where((x >= thresholds[3]) & (x < thresholds[4]), h4, latent)
              latent = np.clip(latent, 0, 1)
