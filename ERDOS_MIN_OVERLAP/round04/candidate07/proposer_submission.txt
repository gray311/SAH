      def run(ctx, args):
          import numpy as np
          N = 800
          domain = 2.0
          dx = domain / N
          
          # Define search grid
          lrs = [0.005, 0.01, 0.02, 0.05, 0.015]
          penalties = [500, 1000, 2000, 3000, 5000]
          
          # Light probe of each combination with 1 seed
          results = []
          for lr, penalty in zip(lrs, penalties):
              # Create a quick test h using seed's bimodal pattern
              x = np.linspace(0, domain, N)
              a1, a2 = 0.25, 0.75
              bw = 0.15
              latent = np.exp(-((x-a1)/bw)**2 * 20) + np.exp(-((x-a2)/bw)**2 * 20)
              
              # Quick objective estimate (without full evaluation)
              h = 1.0 / (1.0 + np.exp(-latent))
              j = 1.0 - h
              h_pad = np.pad(h, (0, N))
              j_pad = np.pad(j, (0, N))
              
              # FFT-based correlation (cheap approximation)
              corr_fft = np.fft.fft(h_pad) * np.conj(np.fft.fft(j_pad))
              corr = np.fft.ifft(corr_fft).real
              
              c5_bound = np.max(corr * dx)
              # Quick "penalized" bound estimate
              int_h = np.sum(h) * dx
              constraint_term = (int_h - 1.0) ** 2
              
              # Estimate effectiveness (lower bound is better)
              combined = c5_bound / 0.38092303510845016
              results.append({
                  "lr": lr,
                  "penalty": penalty,
                  "c5_raw": c5_bound,
                  "combined": combined,
                  "rank": -combined  # lower is better for sorting
              })
          
          # Sort by c5_raw (lower is better)
          results.sort(key=lambda x: x["c5_raw"])
          
          # Return top candidates with suggestions
          suggestions = []
          for i, r in enumerate(results[:10]):
              suggestions.append({
                  "combination": i+1,
                  "lr": r["lr"],
                  "penalty": r["penalty"],
                  "estimated_c5": r["c5_raw"],
                  "suggested_num_restarts": max(8, 3 + i),
                  "suggested_num_steps": min(80000, 50000 + i * 10000)
              })
          
          return {"suggestions": suggestions, "best_c5": results[0]["c5_raw"] if results else float("inf")}
