      def run(ctx, args):
          import jax
          import jax.numpy as jnp
          import optax
          
          # Default parameters
          lr = args.learning_rate if 'learning_rate' in args else 0.01
          penalty = args.penalty_strength if 'penalty_strength' in args else 1000
          steps = args.num_steps if 'num_steps' in args else 10000
          restarts = args.num_restarts if 'num_restarts' in args else 2
          use_adamw = args.use_adamw if 'use_adamw' in args else False
          seed = args.initial_seed if 'initial_seed' in args else 42
          
          N = 200  # Reduced discretization for speed
          dx = 2.0 / N
          
          def init_latent(seed, N):
              key = jax.random.PRNGKey(seed)
              latent = jax.random.normal(key, (N,)) * 0.5
              return latent
          
          best_obj = jnp.inf
          best_h = None
          best_integral = None
          
          for restart in range(restarts):
              key = jax.random.PRNGKey(jax.random.PRNGKey(seed) + restart)
              latent = init_latent(seed, N)
              
              optimizer = optax.adamw(lr) if use_adamw else optax.adam(lr)
              opt_state = optimizer.init(latent)
              
              @jax.jit
              def train_step(latent, opt_state):
                  def loss_fn(v):
                      h = jax.nn.sigmoid(v)
                      j_complement = 1.0 - h
                      h_pad = jnp.pad(h, (0, N))
                      j_pad = jnp.pad(j_complement, (0, N))
                      corr = jnp.fft.ifft(jnp.fft.fft(h_pad) * jnp.conj(jnp.fft.fft(j_pad))).real
                      c5 = jnp.max(corr * dx)
                      integral = jnp.sum(h) * dx
                      constraint = (integral - 1.0) ** 2
                      return c5 + penalty * constraint
                  val, grad = jax.value_and_grad(loss_fn)(latent)
                  updates, opt_state = optimizer.update(grad, opt_state)
                  return optax.apply_updates(latent, updates), opt_state
              
              latent, opt_state = latent, opt_state
              for _ in range(steps):
                  latent, opt_state = train_step(latent, opt_state)
                  h = jax.nn.sigmoid(latent)
                  integral = jnp.sum(h) * dx
                  if jnp.abs(integral - 1.0) < 0.5:  # Feasible
                      h_pad = jnp.pad(h, (0, N))
                      j_pad = jnp.pad(1.0 - h, (0, N))
                      corr = jnp.fft.ifft(jnp.fft.fft(h_pad) * jnp.conj(jnp.fft.fft(j_pad))).real
                      obj = jnp.max(corr * dx)
                      if obj < best_obj:
                          best_obj = obj
                          best_h = h.copy()
                          best_integral = float(integral)
          
          result = {"feasible": best_h is not None, 
                    "best_c5": float(best_obj) if best_h is not None else float('inf'),
                    "integral": float(best_integral) if best_h is not None else 0.0,
                    "seed": seed,
                    "note": "Quick feasibility probe - use best config for full eval"}
          return result
