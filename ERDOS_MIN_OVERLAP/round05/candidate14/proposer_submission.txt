      Generate a valid step function h: [0,2]→[0,1] with exactly ∫h=1.0.
      Specify num_steps (2-10), structure_type (uniform/clustered/periodic/asymmetric/custom),
      and optional custom value assignments. Returns code snippet defining the function

      Call this to create candidate functions, then use probe_solution to rank them.
    input_schema:

      properties:
        num_steps:
          type: integer
          minimum: 2
          maximum: 10
        structure_type:

          enum: [uniform, clustered, periodic, asymmetric, custom]

    implementation_py: |
      def run(ctx, args):
          import math
          num_steps = args.get("num_steps", 4)
          struct = args.get("structure_type", "uniform")
          values = args.get("values", None)
          
          N = 800
          domain = 2.0
          dx = domain / N
          
          # Define breakpoint positions and values
          if struct == "uniform":
              positions = [i / num_steps for i in range(num_steps + 1)]
              if values is None:
                  total = num_steps
                  assigned = [1.0 / total for _ in range(num_steps)]
              else:
                  assigned = values[:num_steps]
              # Normalize to integral = 1
              norm = sum(assigned) * dx * domain
                  assigned = [v / norm for v in assigned]
          elif struct == "clustered":
              positions = [0.25, 0.5, 0.75, 1.0]
              positions = sorted(positions[:num_steps + 1])
              if values is None:
                  # Most mass in one region
                  assigned = [1.0 / (3 * dx) / 0.5, 0, 0]  # 3 intervals, normalize
                  # Re-normalize
                  total = sum([v * dx * domain for v in assigned])
                  assigned = [v / total for v in assigned]
              else:
                  assigned = values
              norm = sum([v * ((positions[i+1]-positions[i]) if i < len(positions)-1 else 2.0-positions[-1]) 
                          for i in range(num_steps)])
                  if norm > 0:
                      assigned = [v / norm for v in assigned]
              positions = [0.0] + positions
              positions = sorted(positions)[:num_steps+1]
          elif struct == "periodic":
              positions = [i / num_steps * 2 for i in range(num_steps + 1)]
              if values is None:
                  assigned = [0.5, 0.5] * (num_steps // 2 + 1)[:num_steps]
              else:
                  assigned = values
              norm = sum(assigned) * dx * domain
                  assigned = [v / norm for v in assigned]
          elif struct == "asymmetric":
              positions = [0.0, 0.5, 1.0, 2.0]
              positions = sorted(positions[:num_steps + 1])
              if values is None:
                  assigned = [0.0, 0.8, 0.0, 0.2]
                  norm = sum([v * dx * domain for v in assigned])
                  assigned = [v / norm for v in assigned]
              else:
                  assigned = values
                  norm = sum(assigned) * dx * domain
                  assigned = [v / norm for v in assigned]
          else:  # custom
              positions = [i / num_steps * 2 for i in range(num_steps + 1)]
              if values is None:
                  assigned = [0.5] * num_steps
              else:
                  assigned = values
              norm = sum(assigned) * dx * domain
                  assigned = [v / norm for v in assigned]
          
          # Generate code snippet
          lines = [f"h = jnp.zeros({N})", f"pos_arr = jnp.array({[f'{p:.6f}' for p in positions]})",
                   f"for i in range(len(pos_arr)-1):"]
          for i in range(len(positions)-1):
              lines.append(f"  h[{int(pos_arr[i] / dx)}:{int(pos_arr[i+1] / dx)}] = {assigned[i]:.6f}")
          lines.append("return h")
          code = "\n".join(lines)
          
          return {"code_snippet": code, "num_steps": num_steps,
                  "integral_check": True, "adjusted_values": assigned}
