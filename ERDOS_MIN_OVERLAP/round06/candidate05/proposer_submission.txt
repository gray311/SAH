      Build an explicit step function candidate with controlled breakpoints and values.
      Specify: breakpoints (list of x-coordinates dividing [0,2]), values (list of h-values for each segment).
      Automatically normalizes to satisfy ∫h=1 by scaling values proportionally.
      Returns the normalized function ready for evaluation.
      Use this to create strategic candidates (single-step, double-step, concentrated mass).
    input_schema:
      type: object
      properties:
        breakpoints:
          type: array
          items:
            type: number
          description: x-coordinates where h changes value (including 0 and 2)
        values:
          type: array
          items:
            type: number
          description: h-value for each segment (before normalization)
    implementation_py: |
      def run(ctx, args):
          breakpoints = args.get("breakpoints", [0, 0.5, 1, 1.5, 2])
          values = args.get("values", [1, 0, 0, 0, 0])
          
          # Clip values to [0,1]
          values = jnp.clip(jnp.array(values), 0, 1)
          
          # Normalize to integral = 1
          integral = 0.0
          for i in range(len(breakpoints)-1):
              width = breakpoints[i+1] - breakpoints[i]
              integral += values[i] * width
          
          if integral == 0:
              return {"error": "Integral is zero, cannot normalize"}
          
          scale = 1.0 / integral
          values = jnp.array(values) * scale
          
          # Build the piecewise constant function representation
          num_intervals = 800
          x = jnp.linspace(0, 2, num_intervals)
          h = jnp.zeros(num_intervals)
          
          for i, bp in enumerate(breakpoints[:-1]):
              start_idx = int(bp * (num_intervals - 1))
              end_idx = int((breakpoints[i+1] if i+1 < len(breakpoints)-1 else 2.0) * (num_intervals - 1))
              h[start_idx:end_idx] = values[i]
          
          return {
              "breakpoints": breakpoints,
              "values": values.tolist(),
              "integral": float(integral),
              "normalized": True,
              "h_sample": h[::max(1, num_intervals//100)].tolist()  # light sampling for display
          }
