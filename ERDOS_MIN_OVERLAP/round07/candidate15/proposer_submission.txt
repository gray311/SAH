      def run(ctx, args):
          import jax.numpy as jnp
          structure = args.get("structure", "uniform")
          num_intervals = args.get("num_intervals", 100)
          candidates = []
          if structure == "uniform":
              code = "h = jnp.full((" + str(num_intervals) + "), 0.5)"
          elif structure == "single":
              half = num_intervals // 2
              code = "h = jnp.concatenate([jnp.ones(" + str(half) + "), jnp.zeros(" + str(num_intervals - half) + "))]"
          elif structure == "double":
              mid = num_intervals // 2
              half = mid // 2
              code = "h = jnp.concatenate([jnp.ones(" + str(half) + "), jnp.zeros(" + str(mid) + "), jnp.ones(" + str(half) + "))]"
          elif structure == "concentrated":
              n_half = num_intervals // 2
              code = "h = jnp.concatenate([jnp.full(" + str(n_half) + ", 2.0), jnp.zeros(" + str(num_intervals - n_half) + "))]"
          elif structure == "symmetric":
              n = num_intervals // 4
              code = "h = jnp.zeros(" + str(num_intervals) + ").at[:" + str(n) + "].set(4.0/3.0).at[" + str(n) + "*2:" + str(n) + "*3"].set(4.0/3.0).at[" + str(n) + "*3:" + str(4*n) + "].set(4.0/3.0)"
          candidates.append("num_intervals = " + str(num_intervals) + "\n" + code)
          return {"candidates": candidates, "note": "Generated candidates for structure " + structure}
