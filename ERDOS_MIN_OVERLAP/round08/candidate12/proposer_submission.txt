      def run(ctx, args):
          import math
          import jax.numpy as jnp
          num_intervals = args.get("num_intervals", 200)
          N = num_intervals
          params = args.get("params", {})
          pattern = args.get("pattern", "single_step")
          
          dx = 2.0 / N
          h = jnp.zeros(N)
          
          # Single step: h=1 on [0,1], need to normalize for integral=1
          if pattern == "single_step":
              h = jnp.ones(N // 2)
          
          elif pattern == "double_step":
              half_N = N // 4
              h = jnp.ones(half_N)
              h = h.at[3*half_N:].set(1.0)
          
          elif pattern == "symmetric_3step":
              third_N = N // 3
              h = jnp.zeros(N)
              h = h.at[:third_N].set(1.0/3.0)
              h = h.at[2*third_N:3*third_N].set(1.0/3.0)
          
          elif pattern == "concentrated_mass":
              start_idx = int(0.5 / 2.0 * N)
              end_idx = int(1.5 / 2.0 * N)
              h = h.at[start_idx:end_idx].set(1.0)
          
          elif pattern == "triangular":
              x = jnp.linspace(0, 2, N)
              h = jnp.maximum(0.0, 1.0 - jnp.abs(x - 1.0))
          
          elif pattern == "sine_wave":
              x = jnp.linspace(0, 2, N)
              h = 0.5 + 0.5 * jnp.sin(math.pi * x)
          
          elif pattern == "piecewise_constant":
              h = jnp.full(N, 0.5)
          
          # Normalize h to ensure integral = 1
          integral_h = jnp.sum(h) * dx
          if integral_h > 0 and integral_h != 1.0:
              h = h / integral_h
              h = jnp.clip(h, 0.0, 1.0)
              integral_h = jnp.sum(h) * dx
              if integral_h > 0:
                  h = h / integral_h
          
          # Compute c5_bound
          j_arr = 1.0 - h
          h_padded = jnp.pad(h, (0, N))
          j_padded = jnp.pad(j_arr, (0, N))
          corr_fft = jnp.fft.fft(h_padded) * jnp.conj(jnp.fft.fft(j_padded))
          correlation = jnp.fft.ifft(corr_fft).real
          c5_bound = jnp.max(correlation) * dx
          
          return {
              "h": h.tolist(),
              "c5_bound": float(c5_bound),
              "pattern": pattern,
              "num_intervals": N,
              "integral": float(jnp.sum(h) * dx)
          }
