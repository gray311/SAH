      def run(ctx, args):
          import jax.numpy as jnp
          N = ctx.hypers.num_intervals
          dx = ctx.domain_width / N
          candidates = []
          h1 = jnp.zeros(N)
          h1[:int(1.0/dx)] = 1.0
          candidates.append(("single_block", h1))
          h2 = jnp.ones(N) * 0.5
          candidates.append(("uniform", h2))
          h3 = jnp.zeros(N)
          center = N // 2
          bw = int(0.4 * 2.0 / dx)
          h3[center-bw:center+bw] = 0.5
          candidates.append(("two_bumps", h3))
          h4 = jnp.zeros(N)
          bs = N // 3
          h4[:bs] = 1/3
          h4[bs:2*bs] = 1/3
          h4[2*bs:] = 1/3
          candidates.append(("three_blocks", h4))
          h5 = jnp.zeros(N)
          bw = int(0.25 * 2.0 / dx)
          h5[N//2 - bw:N//2 + bw] = 1.0 / (2*bw*dx)
          candidates.append(("center_concentrated", h5))
          return {"candidates": [(name, h) for name, h in candidates], "count": len(candidates)}
