    Generate 5-8 diverse step function initializations for Erdos problem.
    Returns dict with pattern names mapping to latent vectors.
    Each latent vector when passed through sigmoid gives valid h with values in [0,1].
    Patterns include: bimodal_tight, bimodal_wide, triangular_3step, periodic_2,

  implementation_py: |
    def run(ctx, args):
        import numpy as np
        N = 800
        domain = 2.0
        dx = domain / N
        rng = np.random.default_rng(ctx.budget_left())
        patterns = {}
        x = np.linspace(0, domain, N)

        # Pattern 1: bimodal_tight - narrow peaks at 0.25, 0.75
        a1, a2 = 0.25, 0.75
        bw1, bw2 = 0.12, 0.12
        lat = np.exp(-((x-a1)/bw1)**2 * 25) + np.exp(-((x-a2)/bw2)**2 * 25)
        lat = (lat - np.min(lat)) / (np.max(lat) - np.min(lat) + 1e-10)
        patterns['bimodal_tight'] = lat

        # Pattern 2: bimodal_wide - broader overlapping peaks
        a1, a2 = 0.25, 0.75
        bw1, bw2 = 0.25, 0.25
        lat = np.exp(-((x-a1)/bw1)**2 * 15) + np.exp(-((x-a2)/bw2)**2 * 15)
        lat = (lat - np.min(lat)) / (np.max(lat) - np.min(lat) + 1e-10)
        patterns['bimodal_wide'] = lat

        # Pattern 3: triangular_3step - 3-level linear ramps
        phases = np.array([0.0, 0.333, 0.666])
        levels = np.array([-8, -2, 8])
        lat = np.zeros(N)
        for p, l in zip(phases, levels):
            in_range = (x >= p) & (x < p + 0.333)
            lat = lat + l * in_range
        lat = (lat - np.min(lat)) / (np.max(lat) - np.min(lat) + 1e-10)
        patterns['triangular_3step'] = lat

        # Pattern 4: periodic_2 - alternating high/low
        periodic = 2.0 * (x < 0.5) - 1.0
        lat = periodic * 5.0
        lat = (lat - np.min(lat)) / (np.max(lat) - np.min(lat) + 1e-10)
        patterns['periodic_2'] = lat

        # Pattern 5: periodic_4 - four-cycle pattern
        periodic_4 = np.sin(2 * np.pi * x) + 1.0
        lat = (periodic_4 - np.min(periodic_4)) / (np.max(periodic_4) - np.min(periodic_4) + 1e-10)
        patterns['periodic_4'] = lat

        # Pattern 6: golomb_5 - peaks at optimal spacing
        marks = np.array([0.0, 0.25, 0.625, 0.9375, 1.0])
        kws = np.array([0.08, 0.1, 0.09, 0.09, 0.1])
        lat = np.zeros(N)
        for m, kw in zip(marks, kws):
            lat = lat + 8.0 * np.exp(-((x-m)/kw)**2 * 20)
        lat = (lat - np.min(lat)) / (np.max(lat) - np.min(lat) + 1e-10)
        patterns['golomb_5'] = lat

        # Pattern 7: shifted_bimodal - left-shifted
        a1, a2 = 0.15, 0.55
        bw1, bw2 = 0.12, 0.12
        lat = np.exp(-((x-a1)/bw1)**2 * 25) + np.exp(-((x-a2)/bw2)**2 * 25)
        lat = (lat - np.min(lat)) / (np.max(lat) - np.min(lat) + 1e-10)
        patterns['shifted_bimodal'] = lat

        # Pattern 8: mixed - random combination
        seed_val = int(ctx.budget_left() * 1000)
        rng = np.random.RandomState(seed_val)
        rng_idx = rng.choice([0, 1, 2, 3, 4, 5], size=3, replace=False)
        mixed_seeds = [int(i) for i in rng_idx]
        lat = np.zeros(N)
        for i in mixed_seeds:
            if i == 0:
                lat += np.exp(-((x-0.25)/0.2)**2 * 15)
            elif i == 1:
                lat += np.exp(-((x-0.75)/0.2)**2 * 15)
            elif i == 2:
                lat += 2.0 * (x < 0.5) - 1.0
            elif i == 3:
                lat += np.sin(4 * np.pi * x) + 1.0
            elif i == 4:
                a1, a2 = 0.2, 0.8
                lat += np.exp(-((x-a1)/0.15)**2 * 20) + np.exp(-((x-a2)/0.15)**2 * 20)
            elif i == 5:
                lat += np.exp(-((x-0.4)/0.1)**2 * 25) + np.exp(-((x-0.6)/0.1)**2 * 25)
        lat = (lat - np.min(lat)) / (np.max(lat) - np.min(lat) + 1e-10)
        patterns['mixed'] = lat

        return {'patterns': patterns, 'num_patterns': len(patterns)}
