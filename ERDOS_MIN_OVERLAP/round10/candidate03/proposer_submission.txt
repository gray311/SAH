      def run(ctx, args):
          import numpy as np
          N = 600
          domain = 2.0
          rng = np.random.default_rng(123)
          constructions = {}
          x = np.linspace(0, domain, N)
          
          # bimodal_2peak: Two peaks at 0.25 and 0.75
          a1, a2 = 0.25, 0.75
          bw1, bw2 = 0.12, 0.12
          bimodal = np.zeros(N)
          bimodal[a1-bw1:a1+bw1] = 10.0
          bimodal[a2-bw2:a2+bw2] = 10.0
          bimodal = bimodal + 0.1
          constructions['bimodal_2peak'] = bimodal / np.max(bimodal) * 15.0
          
          # triangular_3step: Three-level step pattern
          t3step = np.zeros(N)
          phases = np.array([0.0, 0.333, 0.666])
          levels = np.array([0, 5, -5])
          for p, l in zip(phases, levels):
              in_range = (x >= p) & (x < p + 0.333)
              t3step = t3step + l * in_range.astype(float)
          constructions['triangular_3step'] = t3step
          
          # golomb_5peak: Five peaks at optimal spacing
          marks = np.array([0.0, 0.4167, 0.8333, 1.25, 1.6667])
          golomb = np.zeros(N)
          for m in marks:
              width = 0.1
              golomb = golomb + 8.0 * np.exp(-((x-m)/width)**2 * 30)
          constructions['golomb_5peak'] = golomb / np.max(golomb) * 20.0
          
          # periodic_phase_shift: Asymmetric pattern
          periodic = np.zeros(N)
          for phase in [0.0, 0.25, 0.5, 0.75]:
              wave = 2.0 * (x >= phase) - 1.0
              periodic = periodic + wave * 3.0
          constructions['periodic_phase_shift'] = periodic
          
          return {"constructions": constructions}
