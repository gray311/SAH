    def run(ctx, args):
        import numpy as np
        N = 800
        domain = 2.0
        rng = np.random.default_rng(123)
        constructions = {}
        x = np.linspace(0, domain, N)
        golomb_scaled = np.array([0.0, 0.5, 1.6, 2.81, 3.0]) / 2.81 * 2.0
        latent = np.zeros(N)
        for mark in golomb_scaled[:5]:
            latent = latent + 6.0 * np.exp(-((x - mark) / 0.1) ** 2 * 20)
        latent = latent + rng.normal(size=N) * 0.15
        constructions['golomb'] = latent
        latent = np.zeros(N)
        for pos in [0.2, 0.5, 0.8]:
            latent = latent + 6.0 * np.exp(-((x - pos) / 0.12) ** 2 * 18)
        constructions['asym_3peak'] = latent
        latent = 3.0 * np.sin(np.pi * x) + 2.0 * np.sin(2 * np.pi * x) + 1.0 * np.sin(3 * np.pi * x)
        latent = latent + rng.normal(size=N) * 0.2
        constructions['harmonic'] = latent
        return {"constructions": constructions, "keys_used": list(constructions.keys())}
