      def run(ctx, args):
          best = ctx.get_best_program()
          if not best:
              return {"note": "no best program yet, call evaluate_solution first"}
          
          current_score = ctx.best_score()
          
          try:
              # Attempt to refine peak positions in the EVOLVE-BLOCK
              import re
              
              # Try shifting peak positions by small amounts
              deltas = [-0.05, -0.03, -0.02, 0.02, 0.03, 0.05]
              
              for delta in deltas:
                  # Look for common peak position patterns
                  if "0.25" in best:
                      new_best = re.sub(r'(0\.25)([,\s])', lambda m: m.group(1) + str(delta) + m.group(2), best)
                      if new_best != best:
                          ctx.stage_edit(new_best)
                          probe = ctx.probe(subsample=2000)
                          if probe and probe.get("c5_bound", 999) < current_score:
                              return {"improvement_found": True, "delta": delta, "probe_score": probe}
                  if "0.75" in best:
                      new_best = re.sub(r'(0\.75)([,\s])', lambda m: m.group(1) + str(delta) + m.group(2), best)
                      if new_best != best:
                          ctx.stage_edit(new_best)
                          probe = ctx.probe(subsample=2000)
                          if probe and probe.get("c5_bound", 999) < current_score:
                              return {"improvement_found": True, "delta": delta, "probe_score": probe}
                  
                  # Restore original
                  ctx.stage_edit(best)
                  
              # If no improvement from peak shifts, try learning rate adjustments
              lr_matches = re.findall(r'base_learning_rate:\s*([\d.]+)', best)
              if lr_matches:
                  for lr in lr_matches:
                      try:
                          lr_val = float(lr)
                          new_lr = lr_val * 1.05
                          new_best = re.sub(r'base_learning_rate:\s*[\d.]+', f'base_learning_rate: {new_lr:.4f}', best)
                          if new_best != best:
                              ctx.stage_edit(new_best)
                              probe = ctx.probe(subsample=2000)
                              if probe and probe.get("c5_bound", 999) < current_score:
                                  return {"improvement_found": True, "lr": new_lr, "probe_score": probe}
                              ctx.stage_edit(best)
                      except:
                          continue
                  
                  ctx.stage_edit(best)
              
              return {"status": "search_completed", "best_score": current_score}
          
          except Exception as e:
              return {"error": str(e), "status": "failed"}
