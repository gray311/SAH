    def run(ctx, args):
        import jax
        import jax.numpy as jnp
        import optax
        from dataclasses import dataclass

        latent = jnp.array(args["latent_h"])
        seed_start = args.get("seed_start", 0)
        num_steps = args.get("num_steps", 3000)
        base_lr = args.get("base_lr", 0.1)
        penalty_strength = args.get("penalty_strength", 50.0)

        # Simulate the optimization loop with warm strong training then annealing
        optimizer = optax.adam(base_lr)
        opt_state = optimizer.init(latent)

        @jax.jit
        def objective_fn(h):
            j = 1.0 - h
            N = 800
            dx = 2.0 / N
            h_padded = jnp.pad(h, (0, N))
            j_padded = jnp.pad(j, (0, N))
            corr_fft = jnp.fft.fft(h_padded) * jnp.conj(jnp.fft.fft(j_padded))
            correlation = jnp.fft.ifft(corr_fft).real
            c5_bound = jnp.max(correlation * dx)

            integral_h = jnp.sum(h) * dx
            constraint_loss = (integral_h - 1.0) ** 2
            return c5_bound + penalty_strength * constraint_loss

        # Phase 1: Strong optimization
        for step in range(num_steps):
            loss, grads = jax.value_and_grad(objective_fn)(latent)
            updates, opt_state = optimizer.update(grads, opt_state)
            latent = optax.apply_updates(latent, updates)

        h = jax.nn.sigmoid(latent)
        j = 1.0 - h
        N = 800
        dx = 2.0 / N
        h_padded = jnp.pad(h, (0, N))
        j_padded = jnp.pad(j, (0, N))
        corr_fft = jnp.fft.fft(h_padded) * jnp.conj(jnp.fft.fft(j_padded))
        correlation = jnp.fft.ifft(corr_fft).real
        c5_bound = float(jnp.max(correlation * dx))

        integral_h = jnp.sum(h) * dx
        return {"c5_bound": c5_bound, "integral_h": float(integral_h), "status": "refined"}
