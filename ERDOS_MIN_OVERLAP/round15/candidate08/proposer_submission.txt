- name: create_piecewise_init
  description: |
    Create a piecewise constant initialization with specified breakpoints and values.
    The latent is converted to h via sigmoid, then normalized to satisfy integral(h)=1.
    
    Call this to create fundamentally different structures (not Gaussian/sigmoid-based).
    Examples: "3-finger", "bipartite", "multi-peak", "golomb-spaced".
  input_schema:
    type: object
    properties:
      pattern_type:
        type: string
        description: Type of piecewise pattern ("3-finger", "bipartite", "multi-peak", "golomb")
      params:
        type: object
        description: Pattern-specific parameters
  implementation_py: |
    def run(ctx, args):
        import numpy as np
        N = 800
        domain = 2.0
        dx = domain / N
        pattern_type = args.get("pattern_type", "3-finger")
        params = args.get("params", {})
        
        latents = {}
        
        if pattern_type == "3-finger":
            # High on [0,a], low on [a,b], medium on [b,2]
            a = params.get("a", 0.4)
            b = params.get("b", 1.2)
            h_raw = np.zeros(N)
            h_raw[:int(N*a/2)] = 5.0
            h_raw[int(N*a/2):int(N*b/2)] = -3.0
            h_raw[int(N*b/2):] = 2.0
        elif pattern_type == "bipartite":
            # High on [0,a] U [a,2-a], low on [2-a,2]
            a = params.get("a", 0.5)
            h_raw = np.ones(N)
            h_raw[int(N*a/2):int(N*(2-a)/2)] = -2.0
        elif pattern_type == "multi-peak":
            # Multiple narrow peaks
            n_peaks = params.get("n_peaks", 3)
            peak_locs = np.linspace(0.3, 1.7, n_peaks)
            bw = 0.08
            h_raw = np.zeros(N)
            for ploc in peak_locs:
                h_raw += 10.0 * np.exp(-((np.arange(N) - ploc*N/2)/(N*bw/2))**2)
        elif pattern_type == "golomb":
            # Golomb ruler-inspired: marks at 0, 1, 4, 9, 11 (scaled)
            marks = np.array([0.0, 0.5, 1.2, 1.8, 2.0])
            bw = 0.06
            h_raw = np.zeros(N)
            for m in marks:
                h_raw += 8.0 * np.exp(-((np.arange(N) - m*N/2)/(N*bw/2))**2)
        
        # Convert to h in [0,1] via sigmoid, then normalize to integral=1
        latent = np.tanh(h_raw)  # Keep some negative values for diversity
        h = 1.0 / (1.0 + np.exp(-latent))
        h = np.clip(h, 0.01, 0.99)  # Ensure strictly in (0,1)
        
        # Normalize to integral = 1
        current_integral = np.sum(h) * dx
        if current_integral > 0:
            h = h / current_integral
        
        latents[pattern_type] = latent
        return {"patterns": latents, "num_patterns": 1}
