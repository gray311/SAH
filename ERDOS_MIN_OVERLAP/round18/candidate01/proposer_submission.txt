    def _analytical_c5(h: jnp.ndarray, N: int) -> float:
        h_padded = jnp.pad(h, (0, N))
        j_padded = jnp.pad(1.0 - h, (0, N))
        corr = jnp.fft.ifft(jnp.fft.fft(h_padded) * jnp.conj(jnp.fft.fft(j_padded))).real
        dx = 2.0 / N
        return float(jnp.max(corr * dx))
