    description: |
      Compute analytical c5 bound for a step function h using FFT.
      Returns c5_bound = max_k integral h(x)(1-h(x+k))dx.
      Call this after editing a candidate to quickly rank before full evaluation.
      Input: h array (latent values, will be sigmoided internally).
    input_schema:
      type: object
      properties:
        h:
          type: array
          description: latent array (N=800)
    implementation_py: |
      def run(ctx, args):
          import numpy as np
          h_raw = np.array(args.get('h', []))
          N = len(h_raw)
          domain = 2.0
          dx = domain / N
          
          # Sigmoid to get h in [0,1]
          h = 1.0 / (1.0 + np.exp(-h_raw))
          
          # Compute j = 1 - h
          j = 1.0 - h
          
          # Pad to avoid edge effects
          h_padded = np.pad(h, (0, N), mode='constant')
          j_padded = np.pad(j, (0, N), mode='constant')
          
          # FFT-based correlation
          corr_fft = np.fft.fft(h_padded) * np.conj(np.fft.fft(j_padded))
          correlation = np.fft.ifft(corr_fft).real
          
          c5_bound = float(np.max(correlation) * dx)
          
          return {"h": h.tolist(), "c5_bound": c5_bound, "integral": float(np.sum(h) * dx)}
