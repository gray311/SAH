    def run(ctx, args):
        import numpy as np
        N = 800
        domain = 2.0
        dx = domain / N

        def make_pattern(peak_positions, amplitude=4.0, width=N*0.1):
            h = np.zeros(N)
            for peak in peak_positions:
                pos = int(peak * N)
                h += amplitude * np.exp(-((np.arange(N) - pos) ** 2) / (2 * width ** 2))
            h = np.clip(h, 0.001, 5.0)
            h = np.exp(h)
            h = h / (np.sum(h) + 1e-10)
            h = h / (np.sum(h) * dx)
            return np.clip(h, 0.001, 1.0)

        def compute_c5(h):
            h_pad = np.pad(h, (0, N))
            j_pad = np.pad(1.0 - h, (0, N))
            corr = np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(j_pad))).real
            return float(np.max(corr * dx))

        # Candidate 1: Tri-modal (base pattern 14)
        tri_h = make_pattern([0.4, 1.0, 1.6], amplitude=4.0, width=N*0.15)
        tri_c5 = compute_c5(tri_h)
        tri_integral = float(np.sum(tri_h) * dx)

        # Candidate 2: Golomb ruler (5 marks)
        golomb_h = make_pattern([0.0, 0.4, 0.8, 1.2, 1.6], amplitude=3.0, width=N*0.08)
        golomb_c5 = compute_c5(golomb_h)
        golomb_integral = float(np.sum(golomb_h) * dx)

        # Candidate 3: Asymmetric 4-peak
        asym_h = make_pattern([0.3, 0.85, 1.4, 1.9], amplitude=3.5, width=N*0.12)
        asym_c5 = compute_c5(asym_h)
        asym_integral = float(np.sum(asym_h) * dx)

        candidates = [
            {"h": tri_h.tolist(), "integral": tri_integral, "c5_bound": tri_c5, "pattern_type": "tri_modal_base"},
            {"h": golomb_h.tolist(), "integral": golomb_integral, "c5_bound": golomb_c5, "pattern_type": "golomb_5"},
            {"h": asym_h.tolist(), "integral": asym_integral, "c5_bound": asym_c5, "pattern_type": "asym_4peak"}
        ]
        return {"candidates": candidates, "num_candidates": 3}
