    def run(ctx, args):
        import numpy as np
        N = 800
        domain = 2.0
        dx = domain / N
        
        pattern_type = args.get("pattern_type", "auto")
        num_variants = args.get("num_variants", 5)
        
        candidates = []
        
        if pattern_type in ["auto", "bipartite"]:
            # Generate 2-segment (bipartite) patterns
            for a in [0.25, 0.3, 0.33, 0.4, 0.5, 0.6, 0.7]:
                if a >= 1:
                    continue
                h2 = (1 - a) / (2 - a)
                if h2 < 0 or h2 > 1:
                    continue
                
                h = np.zeros(N)
                h[:int(a*N)] = 1.0
                h[int(a*N):] = h2
                
                j = 1.0 - h
                h_padded = np.pad(h, (0, N))
                j_padded = np.pad(j, (0, N))
                corr_fft = np.fft.fft(h_padded) * np.conj(np.fft.fft(j_padded))
                correlation = np.fft.ifft(corr_fft).real
                c5 = np.max(correlation * dx)
                
                if c5 < 0.5:
                    candidates.append({
                        "h": [{"x": 0.0, "v": 1.0}, {"x": a, "v": h2}],
                        "integral": float(np.sum(h) * dx),
                        "c5_bound": float(c5),
                        "pattern_type": f"bipartite_a{a}"
                    })
        
        if pattern_type in ["auto", "trimodal"]:
            # Generate 3-segment symmetric patterns
            for mid in [0.5, 0.6, 0.7]:
                h_left = mid
                h_right = 1.0 - mid
                h_mid = 1.0
                
                h = np.zeros(N)
                h[:int(h_left*N)] = h_left
                h[int(h_left*N):int(mid*N)] = h_mid
                h[int(mid*N):] = h_right
                
                j = 1.0 - h
                h_padded = np.pad(h, (0, N))
                j_padded = np.pad(j, (0, N))
                corr_fft = np.fft.fft(h_padded) * np.conj(np.fft.fft(j_padded))
                correlation = np.fft.ifft(corr_fft).real
                c5 = np.max(correlation * dx)
                
                if c5 < 0.5:
                    candidates.append({
                        "h": [{"x": 0.0, "v": h_left}, {"x": h_left, "v": h_mid}, 
                               {"x": mid, "v": h_right}],
                        "integral": float(np.sum(h) * dx),
                        "c5_bound": float(c5),
                        "pattern_type": f"trimodal_symmetric_mid{mid}"
                    })
        
        if pattern_type in ["auto", "quadrisection"]:
            # Generate 4-segment patterns
            for ratio in [0.4, 0.45, 0.5, 0.55, 0.6]:
                cuts = [ratio, ratio + 0.25, ratio + 0.5, 1.0]
                h = np.zeros(N)
                
                for i in range(3):
                    start = cuts[i]
                    end = cuts[i+1]
                    width = end - start
                    height = (1.0 - ratio) / width
                    h[int(start*N):int(end*N)] = height
                
                j = 1.0 - h
                h_padded = np.pad(h, (0, N))
                j_padded = np.pad(j, (0, N))
                corr_fft = np.fft.fft(h_padded) * np.conj(np.fft.fft(j_padded))
                correlation = np.fft.ifft(corr_fft).real
                c5 = np.max(correlation * dx)
                
                if c5 < 0.5:
                    segments = []
                    for i in range(3):
                        start = cuts[i]
                        end = cuts[i+1]
                        segments.append({"x": start, "v": h[int(start*N)]})
                    segments.append({"x": 1.0, "v": h[int((1-ratio)*N)]})
                    
                    candidates.append({
                        "h": segments,
                        "integral": float(np.sum(h) * dx),
                        "c5_bound": float(c5),
                        "pattern_type": f"quadrisection_ratio{ratio}"
                    })
        
        candidates.sort(key=lambda x: x["c5_bound"])
        
        return {"candidates": candidates[:min(10, len(candidates))], "num_candidates": len(candidates)}
