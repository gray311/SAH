          import numpy as np
          N = 800
          domain = 2.0
          dx = domain / N
          temp = args.get("temperature", 0.5)
          
          candidates = []
          
          # Pattern 1: Two equal rectangular pulses
          cand1 = np.zeros(N)
          left_start = int(0.2 * N)
          left_end = int(0.5 * N)
          right_start = int(1.2 * N)
          right_end = int(1.7 * N)
          width = left_end - left_start
          cand1[left_start:left_end] = 1.0 / width
          cand1[right_start:right_end] = 1.0 / width
          cand1 = np.clip(cand1, 0, 1)
          cand1 = cand1 / np.sum(cand1)
          h1 = np.clip(cand1, 0, 1)
          j1 = 1 - h1
          h1_pad = np.pad(h1, (0, N))
          j1_pad = np.pad(j1, (0, N))
          corr1 = np.fft.fft(h1_pad) * np.conj(np.fft.fft(j1_pad))
          c1 = np.max(np.fft.ifft(corr1).real * dx)
          candidates.append({"h": h1.tolist(), "c5_bound": float(c1), "integral": float(np.sum(h1)*dx), "pattern": "two_peaks"})
          
          # Pattern 2: Three equal narrow peaks (tri-modal)
          cand2 = np.zeros(N)
          centers = [0.4, 1.0, 1.6]
          for c in centers:
              start = int(max(0, (c - 0.15) * N))
              end = int(min(N, (c + 0.15) * N))
              cand2[start:end] = 1.0 / 0.3
          cand2 = np.clip(cand2, 0, 1)
          cand2 = cand2 / np.sum(cand2)
          h2 = np.clip(cand2, 0, 1)
          j2 = 1 - h2
          h2_pad = np.pad(h2, (0, N))
          j2_pad = np.pad(j2, (0, N))
          corr2 = np.fft.fft(h2_pad) * np.conj(np.fft.fft(j2_pad))
          c2 = np.max(np.fft.ifft(corr2).real * dx)
          candidates.append({"h": h2.tolist(), "c5_bound": float(c2), "integral": float(np.sum(h2)*dx), "pattern": "tri_modal"})
          
          # Pattern 3: Golomb-like with 4 marks
          cand3 = np.zeros(N)
          marks = [0.0, 0.5, 1.0, 1.5]
          for m in marks:
              start = int(max(0, (m - 0.12) * N))
              end = int(min(N, (m + 0.12) * N))
              cand3[start:end] = 1.0 / 0.24
          cand3 = np.clip(cand3, 0, 1)
          cand3 = cand3 / np.sum(cand3)
          h3 = np.clip(cand3, 0, 1)
          j3 = 1 - h3
          h3_pad = np.pad(h3, (0, N))
          j3_pad = np.pad(j3, (0, N))
          corr3 = np.fft.fft(h3_pad) * np.conj(np.fft.fft(j3_pad))
          c3 = np.max(np.fft.ifft(corr3).real * dx)
          candidates.append({"h": h3.tolist(), "c5_bound": float(c3), "integral": float(np.sum(h3)*dx), "pattern": "golomb_4"})
          
          # Pattern 4: Asymmetric (more mass on left)
          cand4 = np.zeros(N)
          cand4[:int(1.4 * N)] = 1.0 / 1.4
          cand4 = np.clip(cand4, 0, 1)
          h4 = np.clip(cand4, 0, 1)
          j4 = 1 - h4
          h4_pad = np.pad(h4, (0, N))
