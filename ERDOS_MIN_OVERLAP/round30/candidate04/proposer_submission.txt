              reg: float = 1e-4
              penalty: float = 61.0
          
          class Optimizer:
              def __init__(self, config: Config):
                  self.config = config
                  self.dx = 2.0 / config.num_intervals
                  
              def compute_c5(self, h: jnp.ndarray) -> float:
                  h_padded = jnp.pad(h, (0, config.num_intervals))
                  j_padded = jnp.pad(1.0 - h, (0, config.num_intervals))
                  corr = jnp.fft.ifft(jnp.fft.fft(h_padded) * jnp.conj(jnp.fft.fft(j_padded))).real
                  return float(jnp.max(corr * self.dx))

                  return {"h": h.tolist(), "num_intervals": self.config.num_intervals}
              
              def _apply(self, params: dict) -> dict:

                  h = jnp.clip(h, 0.001, 1.0)
                  h = h / (jnp.sum(h) * self.dx)
                  h = jnp.clip(h, 0.001, 1.0)

                  j_val = 1.0 - h_new

                  j_new_padded = jnp.pad(j_val, (0, self.config.num_intervals))

                  correlation = jnp.fft.ifft(corr_fft).real
