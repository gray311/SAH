    def run(ctx, args):
        import numpy as np
        from typing import Dict, Any
        
        n_intervals = args.get("n_on_intervals", 5)
        positions = args.get("positions", None)
        interval_width = args.get("interval_width", 0.25)
        
        N = 800  # Default from seed program
        domain = 2.0
        dx = domain / N
        
        # Get current best program to extract N if not specified
        current_prog = ctx.get_best_program()
        if current_prog:
            try:
                # Try to extract num_intervals from current program
                import re
                match = re.search(r'num_intervals:\s*(\d+)', current_prog)
                if match:
                    N = int(match.group(1))
                    dx = domain / N
            except:
                pass
        
        # If positions not specified, create evenly spaced positions
        if positions is None:
            # Try to evenly space n_intervals across [0, 2]
            spacing = 2.0 / n_intervals
            positions = [i * spacing for i in range(n_intervals)]
        
        # Create step function
        h = np.zeros(N)
        for pos in positions:
            # Find intervals that overlap with this position
            start_idx = int(pos / dx)
            end_idx = start_idx + int(interval_width / dx)
            if end_idx < N:
                h[start_idx:end_idx] = 1.0
        
        # Normalize to satisfy integral constraint
        total_integral = h.sum() * dx
        if total_integral > 0:
            scale = 1.0 / total_integral
            h = h * scale
        
        # Ensure values are in [0, 1]
        h = np.clip(h, 0.0, 1.0)
        
        # Convert to string format expected by executor
        h_str = "array(" + ", ".join([f"{x:.6f}" for x in h]) + ")"
        
        return {
            "h": h_str,
            "n_on_intervals": n_intervals,
            "positions": positions,
            "interval_width": interval_width,
            "message": f"Generated step function with {n_intervals} ON intervals"
        }
