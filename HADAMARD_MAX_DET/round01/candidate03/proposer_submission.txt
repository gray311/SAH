def bareiss_det(A):
    """Fast integer determinant via Bareiss algorithm."""
    if len(A) == 0:
        return 1
    M = [row[:] for row in A]
    for k in range(len(M) - 1):
        if M[k][k] == 0:
            for i in range(k + 1, len(M)):
                if M[i][k] != 0:
                    M[k], M[i] = M[i], M[k]
                    break
            else:
                return 0
        for i in range(k + 1, len(M)):
            for j in range(k + 1, len(M)):
                num = M[i][j] * M[k][k] - M[i][k] * M[k][j]
                den = M[k - 1][k - 1] if k > 0 else 1
                M[i][j] = num // den
    return M[-1][-1]
