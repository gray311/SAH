      def run(ctx, args):
          program = ctx.get_program()
          try:
              exec(program)
              matrix = construct_hadamard_matrix(n=29)
              import numpy as np
              M = np.array(matrix)
              
              # Check orthogonality (Hadamard property)
              ortho_score = np.mean(np.sum(M @ M.T, axis=1) == 29)
              
              # Det estimate using small samples
              det_estimate = int(np.round(np.linalg.det(M)))
              det_abs = abs(det_estimate)
              
              # Check if it's "Hadamard-like"
              row_norms = np.sum(M**2, axis=1)
              is_hadamard_like = np.allclose(row_norms, 29) and ortho_score > 0.9
              
              return {
                  "matrix_shape": str(M.shape),
                  "det_estimate": det_abs,
                  "orthogonality": float(ortho_score),
                  "is_hadamard_like": is_hadamard_like,
                  "det_sign": "positive" if det_estimate > 0 else "negative" if det_estimate < 0 else "zero"
              }
          except Exception as e:
              return {"error": str(e), "note": "failed to analyze"}
