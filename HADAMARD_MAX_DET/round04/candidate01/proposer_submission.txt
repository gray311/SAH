      def run(ctx, args):
          import numpy as np
          best_matrix = ctx.get_best_program()
          m = np.array(best_matrix, dtype=float)
          n = m.shape[0]
          
          row_norms = np.linalg.norm(m, axis=1)
          if np.any(row_norms == 0):
              row_norms = np.ones(n)
          row_corr = np.abs(np.dot(m, m.T)) / np.outer(row_norms, row_norms)
          row_corr = row_corr - np.eye(n)
          row_corr = row_corr / np.sum(row_corr, axis=1, keepdims=True)
          
          col_corr = np.transpose(row_corr)
          
          top_pairs = []
          for i in range(n):
              for j in range(i+1, n):
                  if row_corr[i, j] > 0.95 or col_corr[i, j] > 0.95:
                      top_pairs.append((i, j, row_corr[i, j], col_corr[i, j]))
          
          top_pairs.sort(key=lambda x: -(x[2] + x[3]))
          
          if not top_pairs:
              top_pairs = [(0, 1, 0.5, 0.5)]
          
          best_pair = top_pairs[0]
          row1, row2, rc, cc = best_pair
          
          if rc > cc:
              mut_type = "row_swap"
          else:
              mut_type = "col_swap"
          
          return {
              "matrix_shape": (n, n),
              "max_correlation": float(max(rc, cc)),
              "mut_type": mut_type,
              "recommended_indices": [row1, row2],
              "confidence": float(1.0 - max(rc, cc)),
              "note": "Swap rows {} and {} or cols {} and {}".format(row1, row2, row1, row2)
          }
