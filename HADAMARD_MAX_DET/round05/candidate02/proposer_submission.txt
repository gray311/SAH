    input_schema:

      properties: {}
    implementation_py: |
      def run(ctx, args):
          import numpy as np
          import random
          n = 29
          residues = {0, 1, 4, 5, 6, 7, 9, 13, 16, 20, 22, 23, 24, 25, 28}
          
          seeds = [42, 123, 456, 789, 2024]
          matrices = []
          
          for seed in seeds:
              rng = random.Random(seed)
              H = np.zeros((n, n), dtype=int)
              for i in range(n):
                  for j in range(n):
                      diff = (i - j) % n
                      H[i, j] = 1 if diff in residues else -1
              matrices.append(H.copy())
          
          return {
              "n": n,
              "quadratic_residues": sorted(list(residues)),
              "seeds_used": seeds,
              "matrices": [m.tolist() for m in matrices]
          }
