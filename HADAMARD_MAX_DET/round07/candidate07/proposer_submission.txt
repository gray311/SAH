    description: |
      Analyze a ±1 matrix structure: compute row/col norms, correlation patterns,
      and suggest perturbation strategies. Call this on the current best to guide next search.
    input_schema:
      type: object
      properties: {}
    implementation_py: |
      def run(ctx, args):
          import numpy as np
          matrix_text = ctx.get_program()
          try:
              matrix = eval(matrix_text)
          except:
              if hasattr(ctx, 'get_best_program'):
                  matrix = eval(ctx.get_best_program())
              else:
                  return {"note": "Cannot parse matrix, use get_program directly"}
          
          matrix = np.array(matrix, dtype=float)
          n = len(matrix)
          
          row_norms = np.linalg.norm(matrix, axis=1)
          col_norms = np.linalg.norm(matrix.T, axis=1)
          
          row_corr = []
          for i in range(n-1):
              corr = np.corrcoef(matrix[i], matrix[i+1])[0,1]
              row_corr.append(corr)
          
          avg_row_corr = np.mean(row_corr)
          if avg_row_corr > 0.8:
              perturbation_suggestion = "High correlation between rows - try perturbation_fraction=0.15"
          elif avg_row_corr > 0.5:
              perturbation_suggestion = "Moderate correlation - try perturbation_fraction=0.10"
          else:
              perturbation_suggestion = "Low correlation - try perturbation_fraction=0.05"
          
          return {
              "n": n,
              "row_norms_mean": float(np.mean(row_norms)),
              "row_norms_std": float(np.std(row_norms)),
              "col_norms_mean": float(np.mean(col_norms)),
              "col_norms_std": float(np.std(col_norms)),
              "avg_row_correlation": float(avg_row_corr),
              "perturbation_suggestion": perturbation_suggestion,
              "recommendations": {
                  "try_perturbation_fraction": 0.15 if avg_row_corr > 0.8 else 0.10,
                  "next_seeds": list(range(0, 500, 100))[:5]
              }
          }
