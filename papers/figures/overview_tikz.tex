% Overview of HarnessRL, drawn in TikZ so it compiles with the paper's fonts.
%
% Layout is on an explicit grid so every arrow is short and nothing crosses a
% box: the top row is the outer loop (synthesize a harness), the bottom row is
% the inner loop (run it), and the two feedback boxes sit directly beneath the
% node they feed, so both return paths are simple vertical arrows.
%
%   ctx ---> phi ---> specs ---> review          (outer loop, top row)
%    ^        ^                    |
%    |        |                    v
%   rat <--- adv <--- scores <--- exec <--- mat  (inner loop, bottom row)
%
% Grey = frozen, blue = the only trained component.
\begin{tikzpicture}[
  font=\scriptsize,
  >={Stealth[length=1.5mm]},
  box/.style     = {draw, rounded corners=1.2pt, align=center, inner sep=2.2pt,
                    minimum height=7mm, text width=17mm, line width=.4pt},
  trained/.style = {box, fill=blue!10,   draw=blue!55!black},
  frozen/.style  = {box, fill=black!8,   draw=black!45},
  proc/.style    = {box, fill=white,     draw=black!55},
  sig/.style     = {box, fill=orange!12, draw=orange!60!black},
  flow/.style    = {->, line width=.5pt, draw=black!70},
  fb/.style      = {->, line width=.5pt, draw=orange!65!black, dashed},
  lbl/.style     = {font=\tiny, inner sep=1.2pt},
]

% ---------- top row: outer loop ----------
\node[proc]    (ctx)  at (0,0)    {task context $c_\tau$\\[-2pt]{\tiny spec $\cdot$ incumbent\\[-3pt] $\cdot$ digest}};
\node[trained] (phi)  at (2.6,0)  {proposer $\pi_\phi$\\[-2pt]{\tiny base ${+}$ LoRA\\[-3pt]\textbf{trained}}};
\node[proc]    (spec) at (5.2,0)  {$K$ harness specs\\[-2pt]{\tiny $z_1,\dots,z_K$}};
\node[proc]    (rev)  at (7.8,0)  {review chain\\[-2pt]{\tiny gates $\cdot$ sandbox\\[-3pt] $\cdot$ parity repair}};

% ---------- bottom row: inner loop ----------
\node[sig]     (rat)  at (0,-2.0)   {ratchet\\[-2pt]{\tiny best program\\[-3pt] ${+}$ harness}};
\node[sig]     (adv)  at (2.6,-2.0) {advantage\\[-2pt]{\tiny gap-normalized,\\[-3pt] RLOO}};
\node[proc]    (sco)  at (5.2,-2.0) {scores $s_k$\\[-2pt]{\tiny best valid\\[-3pt] per rollout}};
\node[frozen]  (exe)  at (7.8,-2.0) {executor $M_0$\\[-2pt]{\tiny \textbf{frozen}, budget $B$}};

% ---------- forward flow ----------
\draw[flow] (ctx)  -- (phi);
\draw[flow] (phi)  -- (spec);
\draw[flow] (spec) -- (rev);
\draw[flow] (rev)  -- node[lbl, right] {materialize $H_k$} (exe);
\draw[flow] (exe)  -- (sco);
\draw[flow] (sco)  -- (adv);

% ---------- feedback: both are short verticals ----------
\draw[fb] (adv) -- node[lbl, right] {update $\phi$ \emph{only}} (phi);
\draw[fb] (sco.south) -- ++(0,-0.42) -| (rat.south);
\draw[fb] (rat) -- node[lbl, left] {next round} (ctx);

% ---------- grouping ----------
\begin{scope}[on background layer]
  \node[fit=(ctx)(rev), draw=black!22, rounded corners=2pt, inner sep=3pt] (ob) {};
  \node[fit=(rat)(exe), draw=black!22, rounded corners=2pt, inner sep=3pt] (ib) {};
\end{scope}
\node[lbl, anchor=west, black!55] at ([yshift=2.4mm]ob.north west)
  {\textbf{outer loop} --- synthesize a harness};
\node[lbl, anchor=west, black!55] at ([yshift=-3.6mm]ib.south west)
  {\textbf{inner loop} --- run it on the frozen executor; no gradient reaches $M_0$};
\end{tikzpicture}
